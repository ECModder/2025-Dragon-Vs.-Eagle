#   Each news is sent on its design date by a hidden event
#   (dve_news_send.N, at the end of this file) scheduled at
#   game start in common/on_actions/DVE_news_2025_on_actions.txt:
#   it reaches every country that exists on that date, so also
#   the ones born later (civil wars) and a player who changed tag.
# ============================================================

# ------------------------------------------------------------
#   Senders. Every country gets the hidden event at the same
#   hour: the first one sends the news to all, the flag stops
#   the others.
# ------------------------------------------------------------

country_event = {
	id = dve_news_send.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_global_flag = dve_news_1_sent } }
			set_global_flag = dve_news_1_sent
			every_country = { news_event = { id = dve_news.1 } }
		}
	}
}
