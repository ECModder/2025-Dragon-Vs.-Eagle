# ============================================================
#   DvE - la guerra civile italiana, il colpo di stato militare
#
#   Scheduled da ITA_civil_war_effect (common/scripted_effects/
#   DVE_ITA_civil_war_effects.txt) con "country_event = { id =
#   dve_itacw.1 days = 2 }": 2 giorni dopo lo scoppio della
#   guerra civile, i vertici militari residui si impongono sul
#   governo che resta - l'FF.AA. (autocracy) prende il potere.
# ============================================================

add_namespace = dve_itacw

# 1: il Comando Interforze prende il potere
country_event = {
	id = dve_itacw.1
	title = dve_itacw.1.t
	desc = dve_itacw.1.d
	picture = GFX_report_event_generic_army
	is_triggered_only = yes

	option = {
		name = dve_itacw.1.a
		set_politics = {
			ruling_party = autocracy
			elections_allowed = no
		}
		add_popularity = { ideology = autocracy popularity = 0.30 }
		create_country_leader = {
			name = "Comando Interforze F.F.A.A."
			desc = "POLITICS_LEADER_DESC"
			picture = "portrait_large_ffaa.dds"
			expire = "2100.1.1"
			ideology = military_dictatorship
		}
		add_stability = -0.05
		add_war_support = 0.05
	}
}

# 2: a human Italy picks the faction to play (from ITA_civil_war_effect).
country_event = {
	id = dve_itacw.2
	title = dve_itacw.2.t
	desc = dve_itacw.2.d
	picture = GFX_report_event_generic_army

	is_triggered_only = yes

	option = {
		name = dve_itacw.2.a
		ai_chance = { factor = 100 }
	}
	option = {
		name = dve_itacw.2.b
		trigger = { country_exists = INT }
		ai_chance = { factor = 0 }
		INT = { change_tag_from = ITA }
	}
	option = {
		name = dve_itacw.2.c
		trigger = { country_exists = SRI }
		ai_chance = { factor = 0 }
		SRI = { change_tag_from = ITA }
	}
	option = {
		name = dve_itacw.2.d
		trigger = { country_exists = RSI }
		ai_chance = { factor = 0 }
		RSI = { change_tag_from = ITA }
	}
	option = {
		name = dve_itacw.2.e
		trigger = { country_exists = LOM }
		ai_chance = { factor = 0 }
		LOM = { change_tag_from = ITA }
	}
	option = {
		name = dve_itacw.2.f
		trigger = { country_exists = TYR }
		ai_chance = { factor = 0 }
		TYR = { change_tag_from = ITA }
	}
}
