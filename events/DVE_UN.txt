add_namespace = dve_un

# 1: the applicant confirms the application. The chance is computed here.
country_event = {
	id = dve_un.1
	title = dve_un.1.t
	desc = dve_un.1.d
	picture = GFX_report_event_generic_conference

	is_triggered_only = yes

	immediate = {
		hidden_effect = { DVE_un_compute_chance_effect = yes }
	}

	option = {
		name = dve_un.1.a
		ai_chance = { factor = 100 }
		custom_effect_tooltip = DVE_un_vote_tt
		hidden_effect = {
			set_country_flag = DVE_un_application_pending
			set_temp_variable = { dve_un_roll = random }
			multiply_temp_variable = { dve_un_roll = 100 }
			if = {
				limit = { check_variable = { dve_un_roll < DVE_un_accept_chance } }
				country_event = { id = dve_un.2 days = 5 }
			}
			else = {
				country_event = { id = dve_un.3 days = 5 }
			}
		}
	}
	option = {
		name = dve_un.1.b
		ai_chance = { factor = 0 }
	}
}
