add_namespace = ROM_iohannis

# ============================================================
#   DvE: the return of Klaus Iohannis (romania_elections, "no clear
#   mandate" path).
#
#   Flow:
#     ROM_interim_mandate  -> 20 days -> ROM_iohannis.1 (the letter)
#     accept               -> ROM_iohannis.3 (he meets the parties)
#     ROM_rerun_the_ballot -> ROM_iohannis.4 (the rerun election, with
#                             Iohannis as the 4th option) instead of
#                             dve_rom.5
#     choose Iohannis      -> ROM_iohannis.2 (world news, fired ONCE)
#                             and the ROM_ioh_victory branch unlocks.
#
#   ROM_iohannis.2 is a MAJOR news event: firing it once from Romania
#   already shows it to every country. It must never be fired inside
#   every_country / every_other_country, or each player sees one copy
#   per country in the world (the old 100+ popups bug).
# ============================================================

# 1. The rumour (mysterious, unnamed)
country_event = {
	id = ROM_iohannis.1
	title = ROM_iohannis.1.t
	desc = ROM_iohannis.1.d
	picture = GFX_report_event_generic_read_write
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		original_tag = ROM
		has_country_flag = ROM_wait
		NOT = {
			OR = {
				has_country_flag = ROM_Nicusor_Dan_wins
				has_country_flag = ROM_Crin_Antonescu_wins
				has_country_flag = ROM_George_Simion_wins
				has_country_flag = ROM_Klaus_Iohannis_wins
				has_country_flag = ROM_dracula_path
				has_country_flag = ROM_iohannis_return_accepted
				has_country_flag = ROM_iohannis_return_rejected
				has_completed_focus = ROM_rerun_the_ballot
				has_completed_focus = ROM_the_sealed_crypt
			}
		}
	}

	# Answer the message
	option = {
		name = ROM_iohannis.1.a
		ai_chance = { factor = 20 }
		set_country_flag = ROM_iohannis_return_accepted
		custom_effect_tooltip = ROM_ioh_unlock_tt
		hidden_effect = {
			country_event = { id = ROM_iohannis.3 days = 10 }
		}
	}

	# Leave the past in the past
	option = {
		name = ROM_iohannis.1.b
		ai_chance = { factor = 80 }
		set_country_flag = ROM_iohannis_return_rejected
		add_stability = -0.02
		add_political_power = 25
		custom_effect_tooltip = ROM_ioh_rejected_tt
	}
}

	}
}

# 4. The rerun election with Iohannis on the ballot.
#    Same title and text as dve_rom.5, options a-c copied from it; the 4th option ("no clear
#    mandate") is replaced by Klaus Iohannis. Fired by ROM_rerun_the_ballot
#    only when the letter from Sibiu was answered.
news_event = {
	id = ROM_iohannis.4
	title = dve_rom.5.t
	desc = dve_rom.5.d
	picture = GFX_news_ROM_2025_elections
	is_triggered_only = yes
	fire_only_once = yes

	trigger = { original_tag = ROM }

	# Nicusor Dan
	option = {
		name = dve_rom.5.a
		ai_chance = { factor = 25 }
		set_country_flag = ROM_Nicusor_Dan_wins
		set_politics = { ruling_party = conservative }
		retire_character = ROM_ilie_bolojan
		recruit_character = ROM_nicusor_dan
		promote_character = ROM_nicusor_dan
		mark_focus_tree_layout_dirty = yes
	}
	# Crin Antonescu
	option = {
		name = dve_rom.5.b
		ai_chance = { factor = 20 }
		set_country_flag = ROM_Crin_Antonescu_wins
		set_politics = { ruling_party = progressive }
		mark_focus_tree_layout_dirty = yes
	}
	# George Simion
	option = {
		name = dve_rom.5.c
		ai_chance = { factor = 25 }
		set_country_flag = ROM_George_Simion_wins
		set_politics = { ruling_party = reactionary }
		mark_focus_tree_layout_dirty = yes
	}
	# Klaus Iohannis (replaces "No candidate wins a clear mandate")
	option = {
		name = ROM_iohannis.4.d
		ai_chance = { factor = 30 }
		set_country_flag = ROM_Klaus_Iohannis_wins
		mark_focus_tree_layout_dirty = yes
		custom_effect_tooltip = ROM_ioh_wins_tt
		hidden_effect = {
			news_event = { id = ROM_iohannis.2 days = 1 }
		}
	}
}
