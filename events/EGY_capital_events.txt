# End of the Suez freeze: progress earned during the freeze is credited now
country_event = {
	id = egypt_capital.102
	hidden = yes
	is_triggered_only = yes
	immediate = {
		clr_country_flag = egypt_capital_freeze
		if = {
			limit = {
				has_country_flag = egypt_capital_active
				check_variable = { egypt_capital_pending > 0 }
			}
			set_temp_variable = { egypt_tmp_gain = egypt_capital_pending }
			set_variable = { egypt_capital_pending = 0 }
			egypt_capital_apply_gain = yes
		}
	}
}
