country_event = {
	id = unitedkingdom.4 #Green Party Leader
	title = unitedkingdom.4.t
	desc = unitedkingdom.4.d
	picture = GFX_fish

	is_triggered_only = yes
	option = {
		name = unitedkingdom.4.a
		if = {
			limit = {
				has_completed_focus = ENG_all_we_need_is_green
			}
			promote_character = ENG_adrian_ramsay

			add_popularity = {
    				ideology = centrist
    				popularity = 0.15
			}
			set_party_name = {
    				ideology = centrist
    				long_name = ENG_centrist_party_greens_long
    				name = ENG_centrist_party_greens
			}
		}
		else_if = {
			limit = {
				has_completed_focus = ENG_back_green_socialism
			}
			set_politics = {
    				ruling_party = progressive
    				elections_allowed = yes
			}
			add_popularity = {
    				ideology = progressive
    				popularity = 0.15
			}	
		}
	}
}
