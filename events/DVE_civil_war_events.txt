# ============================================================
#   DvE - con chi sta Mosca
#
#   Parte da RUS_second_civil_war_effect due ore dopo lo scoppio
#   della guerra civile. Mosca e' in guerra con tutte e tre le
#   fazioni: qui decide se restare sola contro tutti o se fare pace
#   con una e combattere le altre due al suo fianco.
#
#   Chi si sceglie non e' gratis: si adotta il suo partito, il suo
#   leader e la sua ideologia, e si perde l'apparato che si aveva.
# ============================================================

add_namespace = dve_civilwar

country_event = {
	id = dve_civilwar.1
	title = dve_civilwar.1.t
	desc = dve_civilwar.1.d
	picture = GFX_report_event_generic_rebels

	is_triggered_only = yes

	# --- restare il governo di Mosca ---
	option = {
		name = dve_civilwar.1.a
		ai_chance = { factor = 55 }

		# nessuna pace: l'apparato tiene la linea contro tutti e tre
		add_stability = 0.05
		add_war_support = 0.05
		add_political_power = 100
		army_experience = 20
	}

	option = {
		name = dve_civilwar.1.b
		ai_chance = { factor = 15 }

		# Mosca depone le armi con una delle tre e le si mette accanto.
		white_peace = SOV
		SOV = {
			if = {
				limit = { NOT = { is_in_faction = yes } }
				create_faction = "FACTION_AUTHSOC_1"
			}
			add_to_faction = ROOT
			give_military_access = ROOT
		}
		give_military_access = SOV

		# e ne adotta bandiera, partito e organo dirigente
		set_politics = {
			ruling_party = authoritarian_socialism
			elections_allowed = no
		}
		set_party_name = { ideology = authoritarian_socialism name = "KPSU" long_name = "Communist Party of the Soviet Union" }
		create_country_leader = {
			name = "Central Soviet"
			desc = "POLITICS_LEADER_DESC"
			picture = ""
			expire = "1965.1.1"
			ideology = sovietism
		}
		add_popularity = { ideology = authoritarian_socialism popularity = 0.35 }
		add_stability = -0.05
		add_war_support = 0.10
	}

	option = {
		name = dve_civilwar.1.c
		ai_chance = { factor = 15 }

		# Mosca depone le armi con una delle tre e le si mette accanto.
		white_peace = RUR
		RUR = {
			if = {
				limit = { NOT = { is_in_faction = yes } }
				create_faction = "DVE_FACTION_RUR"
			}
			add_to_faction = ROOT
			give_military_access = ROOT
		}
		give_military_access = RUR

		# e ne adotta bandiera, partito e organo dirigente
		set_politics = {
			ruling_party = centrist
			elections_allowed = no
		}
		set_party_name = { ideology = centrist name = "Civic Platform" long_name = "Civic Platform - Constitutional Coalition" }
		create_country_leader = {
			name = "Provisional Assembly"
			desc = "POLITICS_LEADER_DESC"
			picture = ""
			expire = "1965.1.1"
			ideology = social_liberal
		}
		add_popularity = { ideology = centrist popularity = 0.35 }
		add_stability = -0.05
		add_war_support = 0.10
	}

	option = {
		name = dve_civilwar.1.d
		ai_chance = { factor = 15 }

		# Mosca depone le armi con una delle tre e le si mette accanto.
		white_peace = NRE
		NRE = {
			if = {
				limit = { NOT = { is_in_faction = yes } }
				create_faction = "DVE_FACTION_NRE"
			}
			add_to_faction = ROOT
			give_military_access = ROOT
		}
		give_military_access = NRE

		# e ne adotta bandiera, partito e organo dirigente
		set_politics = {
			ruling_party = ultranationalism
			elections_allowed = no
		}
		set_party_name = { ideology = ultranationalism name = "NBP" long_name = "National Bolshevik Party" }
		create_country_leader = {
			name = "Supreme Command"
			desc = "POLITICS_LEADER_DESC"
			picture = ""
			expire = "1965.1.1"
			ideology = national_bolshevism
		}
		add_popularity = { ideology = ultranationalism popularity = 0.35 }
		add_stability = -0.05
		add_war_support = 0.10
	}
}
