#   Senders. Every country gets the hidden event at the same
#   hour: the first one acts for all, the flag stops the others.
# ------------------------------------------------------------

	id = dve_rom_send.1
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { NOT = { has_global_flag = dve_rom_1_sent } }
			set_global_flag = dve_rom_1_sent
			every_country = { news_event = { id = dve_rom.1 } }
		}
	}
}
