# ============================================================
#   DvE - Arms / Surplus / Black Market: delayed delivery
#
#   One event per purchasable item in common/decisions/
#   DVE_market_decisions.txt. Each delivers its equipment 25
#   days after the order was placed and frees the shared
#   DVE_market_orders_active slot back up.
# ============================================================

add_namespace = dve_market

# --- Arms Market ---

	id = dve_market.1
	title = dve_market.1.t
	desc = dve_market.1.d
	picture = GFX_report_event_generic_factory
	is_triggered_only = yes
	option = {
		name = dve_market.1.a
		add_equipment_to_stockpile = { type = infantry_equipment amount = 10000 producer = ROOT }
		add_to_variable = { DVE_market_orders_active = -1 }
	}
}
