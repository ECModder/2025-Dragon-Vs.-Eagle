# DvE government formation - shared triggers.

DVE_gov_has_formateur = {
	OR = {
		has_country_flag = DVE_gov_formateur_totalitarian_socialism
		has_country_flag = DVE_gov_formateur_authoritarian_socialism
		has_country_flag = DVE_gov_formateur_progressive
		has_country_flag = DVE_gov_formateur_liberal
		has_country_flag = DVE_gov_formateur_centrist
		has_country_flag = DVE_gov_formateur_conservative
		has_country_flag = DVE_gov_formateur_reactionary
		has_country_flag = DVE_gov_formateur_authoritarianism
		has_country_flag = DVE_gov_formateur_autocracy
		has_country_flag = DVE_gov_formateur_absolutism
		has_country_flag = DVE_gov_formateur_ultranationalism
	}
}

# At least one party that is allowed to form a government has 20%.
DVE_gov_any_party_can_form = {
	OR = {
		totalitarian_socialism > 0.199
		authoritarian_socialism > 0.199
		progressive > 0.199
		liberal > 0.199
		centrist > 0.199
		conservative > 0.199
		reactionary > 0.199
		authoritarianism > 0.199
		autocracy > 0.199
		absolutism > 0.199
		ultranationalism > 0.199
	}
}
