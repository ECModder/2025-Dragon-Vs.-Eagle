# DvE - European Union: triggers (button states, membership checks, voting rules).

# Candidate with 100 integration, no pending application, no cooldown.
DVE_eu_can_apply = {
	has_country_flag = DVE_eu_candidate
	check_variable = { DVE_eu_integration = 100 }
	NOT = { has_country_flag = DVE_eu_application_pending }
	NOT = { has_country_flag = DVE_eu_application_cooldown }
	has_political_power > 49.999
}

# Application accepted globally - this trigger just checks the flag is set by global resolution.
DVE_eu_can_join = {
	has_country_flag = DVE_eu_application_pending
	has_global_flag = DVE_eu_application_accepted
}

DVE_eu_is_member = { has_country_flag = DVE_eu_member }
DVE_eu_is_candidate = { has_country_flag = DVE_eu_candidate }

DVE_eu_ep_group_ecr = { check_variable = { DVE_eu_ep_delegation_ecr > 0 } }

DVE_eu_ep_group_ni = { check_variable = { DVE_eu_ep_delegation_ni > 0 } }

# QMV passes if 55% states (15) AND 65% population.

	check_variable = { global.DVE_eu_council_qmv_pass = 1 }
}
