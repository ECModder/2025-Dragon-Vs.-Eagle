# ============================================================
#   DvE - SYRIAN CIVIL WAR, March 2025
#   At game start the Interim Government (SYR) holds all of
#   Syria except Rojava (ROJ, untouched). Between 5 and 7 March
#   the rest breaks up and every faction fights every other:
#     SYA  Latakia-Tartus (1070)
#     TSY  Aleppo (1149), Herema Cizire (1135)
#     DRZ  Druze (1150)
#     ISS  Aleppo-Homs (677)
#     SYR  keeps Damascus (554)
#   Scheduled by common/on_actions/DVE_SYR_civil_war_on_actions.txt.
# ============================================================

# Starts the war from anywhere (schedule, debug, safety net).
SYR_civil_war_start_effect = {
	if = {
		limit = {
			country_exists = SYR
			NOT = { has_global_flag = SYR_civil_war }
		}
		SYR = { country_event = { id = dve_syrcw.1 } }
	}
}

SYR_cw_spawn_SYA_effect = {
	every_owned_state = {
		limit = {
			OR = {
				state = 1070 # Latakia-Tartus
			}
		}
		add_core_of = SYA
		SYA = { transfer_state = PREV }
	}
	if = {
		limit = { country_exists = SYA }
		SYA = {
			DVE_cw_new_faction_setup_effect = yes
			if = {
				limit = { owns_state = 1070 }
				set_capital = { state = 1070 }
			}
			SYR_cw_army_SYA_effect = yes
		}
	}
}

# Runs in SYR's scope from dve_syrcw.1 (ROOT = SYR).
SYR_civil_war_effect = {
	set_global_flag = SYR_civil_war
	add_stability = -0.20
	add_war_support = 0.10

	# territory, kit and armies of the four breakaway factions
	SYR_cw_spawn_SYA_effect = yes
	SYR_cw_spawn_TSY_effect = yes
	SYR_cw_spawn_DRZ_effect = yes
	SYR_cw_spawn_ISS_effect = yes
	SYR_cw_army_SYR_effect = yes

	# every faction against every other (Rojava stays out)
	every_country = {
		limit = { DVE_cw_syria = yes }
		every_country = {
			limit = {
				DVE_cw_syria = yes
				NOT = { tag = PREV }
				NOT = { has_war_with = PREV }
				NOT = { is_in_faction_with = PREV }
			}
			PREV = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}

	every_country = {
		limit = { NOT = { DVE_cw_syria = yes } }
		news_event = { id = dve_syrcw.2 hours = 6 }
	}
}
