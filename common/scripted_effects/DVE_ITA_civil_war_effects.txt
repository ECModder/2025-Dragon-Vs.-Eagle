# ============================================================
#   DvE - GUERRA CIVILE ITALIANA A SEI FAZIONI
#
#   Per ora richiamabile SOLO dalle decisioni di DEBUG (common/
#   decisions/DVE_debug.txt, DVE_debug_italy_category) - non e'
#   ancora agganciata a nessun focus. Stesso schema, riga per
#   riga, della guerra civile cinese (DVE_CHN_effects.txt): niente
#   start_civil_war (creerebbe tag dinamici invece di quelli
#   veri), ogni fazione riceve una lista esplicita di stati via
#   every_owned_state + limit, cosi' l'ITA (che sopravvive come
#   fazione a se') mantiene automaticamente tutto cio' che non e'
#   nella lista di nessuno.
#
#   Spartizione (16 stati italiani, tutti assegnati):
#     ITA   4  Abruzzo(157) Sicilia(115) Sardegna(114) Trentino(850)
#     INT   5  Piemonte(158) Lombardia(159) Emilia-Romagna(161)
#              Toscana(162) Trieste(1155)         - capitale 159
#     SRI   3  Napoli/Campania(117) Calabria(156) Puglia(849)
#                                                  - capitale 117
#     RSI   1  Lazio/Roma(2)                       - capitale 2
#     LOM   2  Veneto(160) Bergamo(1088)           - capitale 160
#     TYR   1  Alto Adige(39)                       - capitale 39
#
#   Roma: RSI possiede lo stato per intero (come richiesto), ma
#   l'ITA ci pianta comunque una guarnigione di truppe fedeli
#   (allow_spawning_on_enemy_provs = yes, stesso trucco usato
#   ovunque in questo file per le divisioni create in territorio
#   non proprio) - una sacca di resistenza nella capitale persa,
#   non un possedimento vero.
#
#   Gli stati che l'ITA mantiene per intero (Abruzzo, Sicilia,
#   Sardegna, Trentino) ricevono ognuno 2 divisioni di guarnigione
#   (template "Presidio Territoriale"), da ITA_civil_war_army_effect.
#
#   Guerre dichiarate quando scoppia tutto:
#     ITA contro tutte e cinque le fazioni (il governo legittimo
#       contro chi si e' staccato)
#     INT contro RSI e SRI (comunisti contro fascisti, e le due
#       correnti socialiste rivali - vedi le rispettive history)
#     SRI contro RSI (comunisti contro fascisti)
#     LOM e TYR restano fuori dalle guerre ideologiche: vogliono
#       l'indipendenza, non conquistare il resto del paese.
# ============================================================


# ============================================================
#   INT - Lotta Comunista / Vanguard-Revolutionary Front.
#   5 stati, capitale Milano.
# ============================================================
ITA_cw_spawn_INT_effect = {

	every_owned_state = {
		limit = {
			OR = {
				state = 158  # Piemonte
				state = 159  # Lombardia
				state = 161  # Emilia-Romagna
				state = 162  # Toscana
				state = 1155 # Trieste
			}
		}
		add_core_of = INT
		INT = { transfer_state = PREV }
	}

	if = {
		limit = { country_exists = INT }
		INT = {
			if = {
				limit = { owns_state = 159 }
				set_capital = { state = 159 }
			}
		}
	}
}

# ============================================================
#   LA GUERRA CIVILE COMPLETA
# ============================================================
ITA_civil_war_effect = {

	set_country_flag = ITA_civil_war
	set_country_flag = DVE_civil_war_active

	# --- l'Italia esce dalla NATO: un governo che si sta
	#     sgretolando non puo' restare in un'alleanza militare ---
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}

	add_stability = -0.30
	add_war_support = 0.10

	# --- le cinque fazioni prendono territorio e cores ---
	ITA_cw_spawn_INT_effect = yes
	ITA_cw_spawn_SRI_effect = yes
	ITA_cw_spawn_RSI_effect = yes
	ITA_cw_spawn_LOM_effect = yes
	ITA_cw_spawn_TYR_effect = yes

	# --- eserciti e guarnigione di Roma ---
	ITA_civil_war_army_effect = yes

	# --- 2 giorni dopo, il Comando Interforze F.F.A.A. prende il
	#     potere a Roma (events/DVE_ITA_civil_war.txt) ---
	country_event = { id = dve_itacw.1 days = 2 }

	# --- le guerre: Roma (governo legittimo) contro tutte le fazioni ---
	if = { limit = { country_exists = INT } declare_war_on = { target = INT type = annex_everything } }
	if = { limit = { country_exists = SRI } declare_war_on = { target = SRI type = annex_everything } }
	if = { limit = { country_exists = RSI } declare_war_on = { target = RSI type = annex_everything } }
	if = { limit = { country_exists = LOM } declare_war_on = { target = LOM type = annex_everything } }
	if = { limit = { country_exists = TYR } declare_war_on = { target = TYR type = annex_everything } }

	# --- comunisti contro fascisti ---
	if = {
		limit = { country_exists = INT country_exists = RSI }
		INT = { declare_war_on = { target = RSI type = annex_everything } }
	}
	if = {
		limit = { country_exists = SRI country_exists = RSI }
		SRI = { declare_war_on = { target = RSI type = annex_everything } }
	}

	# --- le due correnti socialiste rivali, come da lore ---
	if = {
		limit = { country_exists = INT country_exists = SRI }
		INT = { declare_war_on = { target = SRI type = annex_everything } }
	}
}

							engineer = { x = 0 y = 0 }

					}

			}

				create_unit = {
					division = "name = \"Guardia Internazionalista\" division_template = \"Guardia Internazionalista\" start_experience_factor = 0.30 start_equipment_factor = 0.90"

# ============================================================
#   ANNULLA TUTTO
#   L'Italia si riprende le cinque fazioni e i loro cores. Serve
#   solo al debug, per rifare la prova da capo.
# ============================================================
ITA_civil_war_reset_effect = {

	clr_country_flag = ITA_civil_war
	clr_country_flag = DVE_civil_war_active

	# --- rientra nella NATO, per rifare la prova da capo ---
	if = {
		limit = { NOT = { is_in_faction = yes } }
		USA = { add_to_faction = ITA }
	}

	if = { limit = { country_exists = INT } INT = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = SRI } SRI = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = RSI } RSI = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = LOM } LOM = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = TYR } TYR = { every_owned_state = { ROOT = { transfer_state = PREV } } } }

	every_owned_state = {
		remove_core_of = INT
		remove_core_of = SRI
		remove_core_of = RSI
		remove_core_of = LOM
		remove_core_of = TYR
	}
}
