# DvE rework: evolution of the three armed-forces National Spirits.
# Every version of a spirit replaces the previous one with swap_ideas, so a
# country never holds two versions of the same spirit. The final versions are
# one per focus tree (nato = romania_elections, uprise = romania_uprise,
# dracula = romania_dracula). If the spirit is missing for any reason (for
# example a console command), the target version is simply added.

# ---- army ----
ROM_army_spirit_tier_2 = {
	if = {
		limit = { has_idea = ROM_army_spirit }
		swap_ideas = { remove_idea = ROM_army_spirit add_idea = ROM_army_spirit_2 }
	}
	else_if = {
		limit = { NOT = { OR = { has_idea = ROM_army_spirit has_idea = ROM_army_spirit_2 has_idea = ROM_army_spirit_nato has_idea = ROM_army_spirit_uprise has_idea = ROM_army_spirit_dracula } } }
		add_ideas = ROM_army_spirit_2
	}
}
