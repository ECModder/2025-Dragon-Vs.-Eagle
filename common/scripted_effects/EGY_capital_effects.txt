egypt_capital_setup = {
	if = {
		limit = { NOT = { has_country_flag = egypt_capital_initialized } }
		set_country_flag = egypt_capital_initialized
		set_country_flag = egypt_capital_active
		set_country_flag = egypt_capital_phase1
		set_variable = { egypt_capital_progress = 74 }
		set_variable = { egypt_capital_active_count = 0 }
		set_variable = { egypt_capital_civ_load = 0 }
		set_variable = { egypt_capital_cg_load = 0 }
		set_variable = { egypt_capital_pending = 0 }
		set_variable = { egypt_capital_events_fired = 0 }
		random_list = {
			30 = { set_variable = { egypt_capital_event_quota = 0 } }
			35 = { set_variable = { egypt_capital_event_quota = 1 } }
			35 = { set_variable = { egypt_capital_event_quota = 2 } }
		}
		add_dynamic_modifier = { modifier = egypt_capital_construction_load }
		egypt_capital_update_bar = yes
		# Rule 2 deadline: 1 January 2032 (2555 days after the 1 January 2025 start)
		country_event = { id = egypt_capital.100 days = 2555 }
	}
}

egypt_capital_start_project = {
	add_to_variable = { egypt_capital_active_count = 1 }
	add_to_variable = { egypt_capital_civ_load = egypt_tmp_civ }
	add_to_variable = { egypt_capital_cg_load = egypt_tmp_cg }
	force_update_dynamic_modifier = yes
}

egypt_capital_finish_project = {
	subtract_from_variable = { egypt_capital_active_count = 1 }
	clamp_variable = { var = egypt_capital_active_count min = 0 }
	subtract_from_variable = { egypt_capital_civ_load = egypt_tmp_civ }
	clamp_variable = { var = egypt_capital_civ_load min = 0 }
	subtract_from_variable = { egypt_capital_cg_load = egypt_tmp_cg }
	clamp_variable = { var = egypt_capital_cg_load min = 0 }
	force_update_dynamic_modifier = yes
	egypt_capital_apply_gain = yes
}

egypt_capital_apply_gain = {
	if = {
		limit = { has_country_flag = egypt_capital_freeze }
		add_to_variable = { egypt_capital_pending = egypt_tmp_gain }
		custom_effect_tooltip = egypt_capital_tt_frozen
	}
	else = {
		add_to_variable = { egypt_capital_progress = egypt_tmp_gain }
		egypt_capital_check_phase = yes
	}
}

egypt_capital_lose_progress = {
	subtract_from_variable = { egypt_capital_progress = egypt_tmp_loss }
	clamp_variable = { var = egypt_capital_progress min = 0 max = 100 }
	egypt_capital_update_bar = yes
}

egypt_capital_check_phase = {
	clamp_variable = { var = egypt_capital_progress min = 0 max = 100 }
	if = {
		limit = { check_variable = { egypt_capital_progress > 99 } }
		if = {
			limit = { has_country_flag = egypt_capital_phase1 }
			clr_country_flag = egypt_capital_phase1
			set_country_flag = egypt_capital_phase2
			set_variable = { egypt_capital_progress = 0 }
			country_event = { id = egypt_capital.10 }
		}
		else_if = {
			limit = { has_country_flag = egypt_capital_phase2 }
			clr_country_flag = egypt_capital_phase2
			set_country_flag = egypt_capital_phase3
			set_variable = { egypt_capital_progress = 0 }
			country_event = { id = egypt_capital.11 }
		}
		else_if = {
			limit = { has_country_flag = egypt_capital_phase3 }
			clr_country_flag = egypt_capital_phase3
			egypt_capital_complete = yes
		}
	}
	egypt_capital_update_bar = yes
}

# progress bar sprite has 51 frames: frame 1 = 0 %, frame 51 = 100 % (2 % steps)
egypt_capital_update_bar = {
	set_variable = { egypt_capital_bar_frame = egypt_capital_progress }
	divide_variable = { egypt_capital_bar_frame = 2 }
	round_variable = egypt_capital_bar_frame
	add_to_variable = { egypt_capital_bar_frame = 1 }
	clamp_variable = { var = egypt_capital_bar_frame min = 1 max = 51 }
}

egypt_capital_complete = {
	set_country_flag = egypt_capital_completed
	clr_country_flag = egypt_capital_active
	set_variable = { egypt_capital_progress = 100 }
	remove_dynamic_modifier = { modifier = egypt_capital_construction_load }
	add_ideas = egypt_new_horizon_of_the_nile
	add_political_power = 150
	add_war_support = 0.15
	capital_scope = {
		add_extra_state_shared_building_slots = 4
		add_building_construction = { type = industrial_complex level = 3 instant_build = yes }
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
	}
	country_event = { id = egypt_capital.20 }
}

egypt_capital_fail = {
	set_country_flag = egypt_capital_failed
	clr_country_flag = egypt_capital_active
	clr_country_flag = egypt_capital_phase1
	clr_country_flag = egypt_capital_phase2
	clr_country_flag = egypt_capital_phase3
	remove_dynamic_modifier = { modifier = egypt_capital_construction_load }
	add_ideas = egypt_desert_facade
	add_political_power = -100
	add_war_support = -0.15
	custom_effect_tooltip = egypt_capital_tt_unrest
	hidden_effect = {
		every_owned_state = {
			limit = { is_core_of = ROOT }
			add_dynamic_modifier = { modifier = egypt_capital_unrest }
		}
	}
}

egypt_capital_register_event = {
	add_to_variable = { egypt_capital_events_fired = 1 }
	set_country_flag = { flag = egypt_capital_event_cooldown days = 365 value = 1 }
}
