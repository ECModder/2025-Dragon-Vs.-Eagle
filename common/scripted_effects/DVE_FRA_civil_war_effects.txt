# ============================================================
#   DvE - GUERRA CIVILE FRANCESE A UNDICI FAZIONI
#
#   Per ora richiamabile SOLO dalle decisioni di DEBUG (common/
#   decisions/DVE_debug.txt, DVE_debug_france_category) - non e'
#   ancora agganciata a nessun focus. Stesso schema, riga per
#   riga, della guerra civile italiana (DVE_ITA_civil_war_effects.
#   txt): niente start_civil_war (creerebbe tag dinamici invece di
#   quelli veri), ogni fazione riceve una lista esplicita di stati
#   via every_owned_state + limit, cosi' la FRA (che sopravvive
#   come Repubblica legittima) mantiene automaticamente tutto cio'
#   che non e' nella lista di nessuno - in questo caso Ile de
#   France/Parigi (16), unico stato non assegnato dei 33.
#
#   Spartizione (33 stati francesi, tutti assegnati):
#     FRA   1  Ile de France(16) - resta alla Repubblica, capitale
#     FSR   2  Loire(30) Poitou(23)             - capitale 30
#     CMU   3  Champagne(18) Champagne2(27) Centre(24) - cap. 24
#     FRS   9  Normandy(15) Alcase Lorraine(17) Alcase(28)
#              Franche-comte(20) Bourgogne(26) Limousin(25)
#              Rhone Alpes(32) Savoy(735) Centre sud(33) - cap. 32
#     MOF   2  Picardy(785) Pas de Calais(29)   - capitale 29
#     BRY   1  Brittany(14)                     - capitale 14
#     VAF   2  Bouches-du-Rhone(21) Var(851)    - capitale 21
#     ANF   4  Aquitaine(19) French Basque Country(806)
#              Midi Pyrenees(31) Roussillion/Languedoc(22) - cap. 19
#     FAC   7  French SA(310) New Caledonia(635) Tahiti(641)
#              French Caribbean(694) Reunion(706) Kerguelen(713)
#              St Pierre and Miquelon(730)       - capitale 694
#     COR   1  Corsica(1)                       - capitale 1
#     CTL   1  Northern Catalonia(1071)          - capitale 1071
#
#   FSR e COR esistevano gia' (rispettivamente dal sistema "Europa
#   Rossa" esportato dall'Italia e come tag autonomo): qui vengono
#   solo riutilizzati con la nuova identita' politica gia' scritta
#   nei rispettivi history/countries/. Vedi il commento in testa a
#   "FSR - French Socialist Republic.txt" per la nota sulla
#   convivenza con ITA_red_france_effect (DVE_ITA_red_europe.txt),
#   che punta allo STESSO tag con un esito politico diverso.
#
#   Guerre dichiarate quando scoppia tutto:
#     FRA (governo legittimo) contro tutte e dieci le fazioni
#     CMU contro FRS (comunisti contro il National Bloc)
#   Le altre fazioni (MOF, BRY, VAF, ANF, FAC, COR, CTL, FSR) non
#   si dichiarano guerra a vicenda: sono rotture separate, non
#   fronti comuni - restano comunque tutte in guerra CONTRO la FRA.
# ============================================================


# ============================================================
#   FSR - Socialist France (Olivier Faure/PS).
#   2 stati, capitale Loire.
# ============================================================
FRA_cw_spawn_FSR_effect = {

	every_owned_state = {
		limit = {
			OR = {
				state = 30  # Loire
				state = 23  # Poitou
			}
		}
		add_core_of = FSR
		FSR = { transfer_state = PREV }
	}

	if = {
		limit = { country_exists = FSR }
		FSR = {
			if = {
				limit = { owns_state = 30 }
				set_capital = { state = 30 }
			}
		}
	}
}

# ============================================================
#   LA GUERRA CIVILE COMPLETA
# ============================================================
FRA_civil_war_effect = {

	set_country_flag = FRA_civil_war
	set_country_flag = DVE_civil_war_active

	# --- la Francia esce dalla NATO: un governo che si sta
	#     sgretolando non puo' restare in un'alleanza militare ---
	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}

	add_stability = -0.35
	add_war_support = 0.10

	# --- Mélenchon si insedia subito: e' gia' il presidente eletto
	#     nel 2027, la guerra civile lo trova gia' al potere ---
	set_politics = {
		ruling_party = liberal
		elections_allowed = no
	}
	create_country_leader = { name = "Jean-Luc Mélenchon" desc = "POLITICS_LEADER_DESC" picture = "" expire = "2035.1.1" ideology = left_populism }
	add_popularity = { ideology = liberal popularity = 0.20 }

	# --- le dieci fazioni prendono territorio e cores ---
	FRA_cw_spawn_FSR_effect = yes
	FRA_cw_spawn_CMU_effect = yes
	FRA_cw_spawn_FRS_effect = yes
	FRA_cw_spawn_MOF_effect = yes
	FRA_cw_spawn_BRY_effect = yes
	FRA_cw_spawn_VAF_effect = yes
	FRA_cw_spawn_ANF_effect = yes
	FRA_cw_spawn_FAC_effect = yes
	FRA_cw_spawn_COR_effect = yes
	FRA_cw_spawn_CTL_effect = yes

	# --- eserciti ---
	FRA_civil_war_army_effect = yes

	# --- le guerre: Parigi (governo legittimo) contro tutte le fazioni ---
	if = { limit = { country_exists = FSR } declare_war_on = { target = FSR type = annex_everything } }
	if = { limit = { country_exists = CMU } declare_war_on = { target = CMU type = annex_everything } }
	if = { limit = { country_exists = FRS } declare_war_on = { target = FRS type = annex_everything } }
	if = { limit = { country_exists = MOF } declare_war_on = { target = MOF type = annex_everything } }
	if = { limit = { country_exists = BRY } declare_war_on = { target = BRY type = annex_everything } }
	if = { limit = { country_exists = VAF } declare_war_on = { target = VAF type = annex_everything } }
	if = { limit = { country_exists = ANF } declare_war_on = { target = ANF type = annex_everything } }
	if = { limit = { country_exists = FAC } declare_war_on = { target = FAC type = annex_everything } }
	if = { limit = { country_exists = COR } declare_war_on = { target = COR type = annex_everything } }
	if = { limit = { country_exists = CTL } declare_war_on = { target = CTL type = annex_everything } }

	# --- comunisti contro il National Bloc ---
	if = {
		limit = { country_exists = CMU country_exists = FRS }
		CMU = { declare_war_on = { target = FRS type = annex_everything } }
	}
}
