# ============================================================
#   DvE - French Civil War, DEBUG-only for now.
#
#   Two waves, matching the Italian civil war's delayed-event
#   pattern (DVE_ITA_civil_war_effects.txt):
#     Day 1: FRS, CMU and COR break away. FRA keeps the rest.
#     Day 7: FRS collapses (BRY takes Brittany, FAC takes the
#            other 9 states), FSR and ANF carve territory out of
#            FRA, and GER occupies Alsace-Lorraine.
#
#   VAF and CTL are NOT part of this territorial mechanic (no
#   state ever assigned to them) - their history/country files
#   stay registered but unused.
#
#   All 33 mainland+overseas states, verified with no gaps or
#   overlaps across both waves:
#     Day 1  FRS 10  CMU 2  COR 1  FRA 20 (incl. 7 overseas)
#     Day 7  MOF 1(from FRA)  BRY 1(from FRS)  FAC 9(from FRS)
#            FSR 2(from FRA)  ANF 3(from FRA)  GER 1(from FRA)
#            FRS 0 (dissolved)  FRA 13 (6 metropolitan + 7 overseas)
# ============================================================


# ============================================================
#   DAY 1
# ============================================================

FRA_cw_spawn_FRS_effect = {
	every_owned_state = {
		limit = {
			OR = {
				state = 22   # Languedoc
				state = 21   # Bouches-du-Rhone
				state = 735  # Savoy
				state = 851  # Var
				state = 32   # Rhone-Alpes
				state = 20   # Rhone
				state = 26   # Auvergne
				state = 25   # Limousin
				state = 14   # Brittany
				state = 15   # Normandy
			}
		}
		add_core_of = FRS
		FRS = { transfer_state = PREV }
	}
	if = {
		limit = { country_exists = FRS }
		FRS = {
			if = { limit = { owns_state = 21 } set_capital = { state = 21 } }
		}
	}
}

# ============================================================
#   DAY 1 - full effect
# ============================================================
FRA_civil_war_effect = {

	set_country_flag = FRA_civil_war
	set_country_flag = DVE_civil_war_active

	if = {
		limit = { is_in_faction = yes }
		leave_faction = yes
	}

	add_stability = -0.35
	add_war_support = 0.10

	# Mélenchon is already the elected president in 2027.
	set_politics = {
		ruling_party = liberal
		elections_allowed = no
	}
	create_country_leader = { name = "Jean-Luc Mélenchon" desc = "POLITICS_LEADER_DESC" picture = "" expire = "2035.1.1" ideology = left_populism }
	add_popularity = { ideology = liberal popularity = 0.20 }

	FRA_cw_spawn_FRS_effect = yes
	FRA_cw_spawn_CMU_effect = yes
	FRA_cw_spawn_COR_effect = yes

	FRA_civil_war_army_effect = yes

	if = { limit = { country_exists = FRS } declare_war_on = { target = FRS type = annex_everything } }
	if = { limit = { country_exists = CMU } declare_war_on = { target = CMU type = annex_everything } }
	if = { limit = { country_exists = COR } declare_war_on = { target = COR type = annex_everything } }
	if = {
		limit = { country_exists = CMU country_exists = FRS }
		CMU = { declare_war_on = { target = FRS type = annex_everything } }
	}

	# Second wave in 7 days: FRS collapses, FSR/ANF/GER carve up the rest.
	country_event = { id = dve_fracw.1 days = 7 }
}

# ============================================================
#   RESET - undoes both waves
# ============================================================
FRA_civil_war_reset_effect = {

	clr_country_flag = FRA_civil_war
	clr_country_flag = DVE_civil_war_active

	if = {
		limit = { NOT = { is_in_faction = yes } }
		USA = { add_to_faction = FRA }
	}

	if = { limit = { country_exists = FRS } FRS = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = CMU } CMU = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = COR } COR = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = MOF } MOF = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = BRY } BRY = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = FAC } FAC = { every_owned_state = { ROOT = { transfer_state = PREV } } } }

	# FSR and ANF get their new (day-7) territory back; GER's
	# occupation of Alsace-Lorraine is undone the same way.
	every_country = {
		limit = { OR = { tag = FSR tag = ANF tag = GER } }
		every_owned_state = {
			limit = {
				OR = {
					state = 29 state = 785 state = 806 state = 31 state = 1071 state = 28
				}
			}
			ROOT = { transfer_state = PREV }
		}
	}

	every_owned_state = {
		remove_core_of = FRS
		remove_core_of = CMU
		remove_core_of = COR
		remove_core_of = MOF
		remove_core_of = BRY
		remove_core_of = FAC
		remove_core_of = FSR
		remove_core_of = ANF
	}
}
