# ============================================================
#   DvE - PORTUGAL (POR) - scripted effects
#
#   Variables (all on POR, all 0-100 unless noted):
#     POR_parl_support    Parliamentary Stability minigame
#     POR_confidence      public confidence (shared by several minigames)
#     POR_camp_north / POR_camp_lisbon / POR_camp_south   Presidential Campaign
#     POR_campaign        campaign momentum, -25..25
#     POR_relief_points   Blackout Recovery: allocations left (starts at 6)
#     POR_bo_essential / POR_bo_services / POR_bo_grid   Blackout allocations
#     POR_renewal_support / _budget / _coop / _ambition   Ronaldo's National Renewal
#     POR_def_recruit / _training / _equipment / _strain  Defence Modernisation
#     POR_luso_partners   Lusophone partnership treaties signed (count)
#     POR_iberian_talks   progress of the Iberian negotiations
#
#   Route flags (exactly one at a time after 9 March 2026):
#     POR_route_montenegro, POR_route_santos, POR_route_ventura, POR_route_ronaldo
#
#   Change a variable, then call its POR_update_* effect: the update
#   clamps the value and keeps exactly one tier spirit, so it is safe
#   to call any number of times.
# ============================================================

POR_init = {
	if = {
		limit = { NOT = { has_variable = POR_parl_support } }
		set_variable = { POR_parl_support = 45 }
		set_variable = { POR_confidence = 50 }
		POR_update_parliament = yes
	}
}

# ---- Parliamentary Stability -------------------------------------------
POR_clear_parliament_tiers = {
	if = { limit = { has_idea = POR_parl_fragile } remove_ideas = POR_parl_fragile }
	if = { limit = { has_idea = POR_parl_working } remove_ideas = POR_parl_working }
	if = { limit = { has_idea = POR_parl_stable } remove_ideas = POR_parl_stable }
}

POR_update_parliament = {
	clamp_variable = { var = POR_parl_support min = 0 max = 100 }
	clamp_variable = { var = POR_confidence min = 0 max = 100 }
	if = {
		limit = { check_variable = { POR_parl_support < 40 } }
		if = { limit = { NOT = { has_idea = POR_parl_fragile } } POR_clear_parliament_tiers = yes add_ideas = POR_parl_fragile }
	}
	else_if = {
		limit = { check_variable = { POR_parl_support < 70 } }
		if = { limit = { NOT = { has_idea = POR_parl_working } } POR_clear_parliament_tiers = yes add_ideas = POR_parl_working }
	}
	else = {
		if = { limit = { NOT = { has_idea = POR_parl_stable } } POR_clear_parliament_tiers = yes add_ideas = POR_parl_stable }
		if = {
			limit = { NOT = { has_country_flag = POR_stable_majority_seen } }
			set_country_flag = POR_stable_majority_seen
			country_event = { id = por.6 days = 1 }
		}
	}
	# A government crisis: below 25, at most once every 300 days.
	if = {
		limit = {
			check_variable = { POR_parl_support < 25 }
			NOT = { has_country_flag = POR_crisis_cooldown }
		}
		set_country_flag = { flag = POR_crisis_cooldown days = 300 value = 1 }
		country_event = { id = por.5 days = 2 }
	}
}

# ---- Presidential Campaign ---------------------------------------------
POR_campaign_start = {
	if = {
		limit = { NOT = { has_country_flag = POR_campaign_active } NOT = { has_country_flag = POR_election_held } }
		set_country_flag = POR_campaign_active
		set_variable = { POR_camp_north = 30 }
		set_variable = { POR_camp_lisbon = 30 }
		set_variable = { POR_camp_south = 30 }
		set_variable = { POR_campaign = 0 }
		POR_update_campaign = yes
	}
}

POR_update_campaign = {
	clamp_variable = { var = POR_camp_north min = 0 max = 100 }
	clamp_variable = { var = POR_camp_lisbon min = 0 max = 100 }
	clamp_variable = { var = POR_camp_south min = 0 max = 100 }
	clamp_variable = { var = POR_campaign min = -25 max = 25 }
	set_variable = { POR_camp_score = POR_camp_north }
	add_to_variable = { POR_camp_score = POR_camp_lisbon }
	add_to_variable = { POR_camp_score = POR_camp_south }
	divide_variable = { POR_camp_score = 3 }
	add_to_variable = { POR_camp_score = POR_campaign }
}

# Election night, 18 January 2026. A player who backed nobody gets the
# default (historical) outcome: Montenegro.
POR_election_resolve = {
	if = {
		limit = { NOT = { has_country_flag = POR_election_held } }
		set_country_flag = POR_election_held
		clr_country_flag = POR_campaign_active
		if = { limit = { NOT = { has_variable = POR_camp_score } } set_variable = { POR_camp_score = 45 } }
		POR_update_campaign = yes
		if = {
			limit = {
				NOT = { has_country_flag = POR_backed_montenegro }
				NOT = { has_country_flag = POR_backed_santos }
				NOT = { has_country_flag = POR_backed_ventura }
			}
			set_country_flag = POR_backed_montenegro
			set_variable = { POR_camp_score = 45 }
		}
		if = {
			limit = { check_variable = { POR_camp_score > 54 } }
			set_country_flag = POR_mandate_strong
		}
		else_if = {
			limit = { check_variable = { POR_camp_score < 40 } }
			set_country_flag = POR_mandate_contested
		}
	}
}

# Sends the winner's news event: next day after a first-round win,
# on 8 February 2026 (21 days later) after a runoff.
POR_send_winner_event = {
	if = {
		limit = { has_country_flag = POR_mandate_strong }
		set_temp_variable = { por_delay = 1 }
	}
	else = {
		set_temp_variable = { por_delay = 21 }
	}
	if = {
		limit = { has_country_flag = POR_backed_santos }
		if = { limit = { check_variable = { por_delay < 2 } } news_event = { id = por.24 days = 1 } }
		else = { news_event = { id = por.24 days = 21 } }
	}
	else_if = {
		limit = { has_country_flag = POR_backed_ventura }
		if = { limit = { check_variable = { por_delay < 2 } } news_event = { id = por.25 days = 1 } }
		else = { news_event = { id = por.25 days = 21 } }
	}
	else = {
		if = { limit = { check_variable = { por_delay < 2 } } news_event = { id = por.23 days = 1 } }
		else = { news_event = { id = por.23 days = 21 } }
	}
}

# Mandate spirit shared by the three winners.
POR_apply_mandate = {
	if = {
		limit = { has_country_flag = POR_mandate_strong }
		add_timed_idea = { idea = POR_strong_mandate days = 365 }
		add_to_variable = { POR_parl_support = 8 }
	}
	else_if = {
		limit = { has_country_flag = POR_mandate_contested }
		add_timed_idea = { idea = POR_contested_mandate days = 365 }
		add_to_variable = { POR_parl_support = -8 }
	}
	POR_update_parliament = yes
}

# Inauguration, 9 March 2026: the route starts here.
POR_inaugurate_winner = {
	if = {
		limit = { NOT = { has_country_flag = POR_inaugurated } }
		set_country_flag = POR_inaugurated
		if = { limit = { has_idea = POR_political_uncertainty } remove_ideas = POR_political_uncertainty }
		if = {
			limit = { has_country_flag = POR_backed_santos }
			set_country_flag = POR_route_santos
			set_politics = { ruling_party = progressive last_election = "2026.1.18" election_frequency = 60 elections_allowed = yes }
			add_ideas = POR_social_renewal_programme
		}
		else_if = {
			limit = { has_country_flag = POR_backed_ventura }
			set_country_flag = POR_route_ventura
			set_politics = { ruling_party = reactionary last_election = "2026.1.18" election_frequency = 60 elections_allowed = yes }
			add_ideas = POR_national_transformation
		}
		else = {
			set_country_flag = POR_route_montenegro
			set_politics = { ruling_party = conservative last_election = "2026.1.18" election_frequency = 60 elections_allowed = yes }
			add_ideas = POR_government_seeking_stability
		}
		POR_apply_mandate = yes
	}
}

# ---- Ronaldo takes office: no candidate-route spirit may survive -------
# Lasting reforms (health service, housing, labour law...) stay: they
# are laws already passed. Only the spirits that describe WHO governs go.
POR_clear_candidate_route = {
	clr_country_flag = POR_route_montenegro
	clr_country_flag = POR_route_santos
	clr_country_flag = POR_route_ventura
	if = { limit = { has_idea = POR_government_seeking_stability } remove_ideas = POR_government_seeking_stability }
	if = { limit = { has_idea = POR_stable_institutions } remove_ideas = POR_stable_institutions }
	if = { limit = { has_idea = POR_social_renewal_programme } remove_ideas = POR_social_renewal_programme }
	if = { limit = { has_idea = POR_social_republic } remove_ideas = POR_social_republic }
	if = { limit = { has_idea = POR_national_transformation } remove_ideas = POR_national_transformation }
	if = { limit = { has_idea = POR_new_political_order } remove_ideas = POR_new_political_order }
	if = { limit = { has_idea = POR_institutional_deadlock } remove_ideas = POR_institutional_deadlock }
	if = { limit = { has_idea = POR_brussels_friction } remove_ideas = POR_brussels_friction }
	if = { limit = { has_idea = POR_strong_mandate } remove_ideas = POR_strong_mandate }
	if = { limit = { has_idea = POR_contested_mandate } remove_ideas = POR_contested_mandate }
}

POR_ronaldo_takes_office = {
	if = {
		limit = { NOT = { has_country_flag = POR_route_ronaldo } }
		POR_clear_candidate_route = yes
		set_country_flag = POR_route_ronaldo
		if = {
			limit = { NOT = { has_character = POR_cristiano_ronaldo } }
			recruit_character = POR_cristiano_ronaldo
		}
		set_politics = { ruling_party = centrist last_election = "2030.5.19" election_frequency = 60 elections_allowed = yes }
		promote_character = POR_cristiano_ronaldo
		if = {
			limit = { has_idea = POR_national_renewal_movement }
			swap_ideas = { remove_idea = POR_national_renewal_movement add_idea = POR_renewal_presidency }
		}
		else = { add_ideas = POR_renewal_presidency }
		add_popularity = { ideology = centrist popularity = 0.20 }
		# hides the election branch (allow_branch on POR_a_republic_at_a_crossroads)
		mark_focus_tree_layout_dirty = yes
	}
}

# ---- Ronaldo's National Renewal ----------------------------------------
POR_renewal_start = {
	if = {
		limit = { NOT = { has_country_flag = POR_ronaldo_movement_founded } }
		set_country_flag = POR_ronaldo_movement_founded
		set_variable = { POR_renewal_support = 35 }
		set_variable = { POR_renewal_budget = 40 }
		set_variable = { POR_renewal_coop = 25 }
		set_variable = { POR_renewal_ambition = 20 }
		set_party_name = { ideology = centrist long_name = "Movimento Renovacao Nacional" name = "MRN" }
		if = {
			limit = { NOT = { has_character = POR_cristiano_ronaldo } }
			recruit_character = POR_cristiano_ronaldo
		}
		add_ideas = POR_national_renewal_movement
		add_popularity = { ideology = centrist popularity = 0.08 }
	}
}

POR_update_renewal = {
	clamp_variable = { var = POR_renewal_support min = 0 max = 100 }
	clamp_variable = { var = POR_renewal_budget min = 0 max = 100 }
	clamp_variable = { var = POR_renewal_coop min = 0 max = 100 }
	clamp_variable = { var = POR_renewal_ambition min = 0 max = 100 }
	# Ambition running 30 points ahead of cooperation frightens the institutions.
	set_temp_variable = { por_gap = POR_renewal_ambition }
	subtract_from_temp_variable = { por_gap = POR_renewal_coop }
	if = {
		limit = {
			check_variable = { por_gap > 30 }
			NOT = { has_country_flag = POR_renewal_backlash_cooldown }
			NOT = { has_country_flag = POR_route_ronaldo }
		}
		set_country_flag = { flag = POR_renewal_backlash_cooldown days = 180 value = 1 }
		country_event = { id = por.67 days = 3 }
	}
}

# ---- Monthly tick (common/on_actions/POR_on_actions.txt) ---------------
POR_monthly = {
	POR_init = yes
	# Parliament: support erodes unless it is maintained.
	if = {
		limit = { NOT = { has_idea = POR_stable_institutions } }
		add_to_variable = { POR_parl_support = -1 }
	}
	if = { limit = { has_idea = POR_contested_mandate } add_to_variable = { POR_parl_support = -1 } }
	if = { limit = { has_idea = POR_institutional_deadlock } add_to_variable = { POR_parl_support = -1 } }
	# Public confidence drifts back towards 50.
	if = { limit = { check_variable = { POR_confidence > 50 } } add_to_variable = { POR_confidence = -1 } }
	else_if = { limit = { check_variable = { POR_confidence < 50 } } add_to_variable = { POR_confidence = 1 } }
	POR_update_parliament = yes

	if = {
		limit = { has_country_flag = POR_ronaldo_movement_founded }
		add_to_variable = { POR_renewal_budget = -2 }
		if = { limit = { check_variable = { POR_renewal_support > 50 } } add_to_variable = { POR_renewal_support = -1 } }
		if = { limit = { check_variable = { POR_renewal_budget < 10 } } add_to_variable = { POR_renewal_support = -3 } }
		POR_update_renewal = yes
	}
	if = {
		limit = { has_country_flag = POR_defence_planning }
		add_to_variable = { POR_def_strain = -3 }
		POR_update_defence = yes
	}
}

			limit = { date > 2029.2.4 date < 2029.3.5 has_country_flag = POR_ronaldo_movement_founded NOT = { has_country_flag = por_66_fired } }
			set_country_flag = por_66_fired
			country_event = { id = por.66 hours = 6 }
		}
		if = {
			limit = { date > 2029.3.8 date < 2029.4.10 NOT = { has_country_flag = por_49_fired } }
			set_country_flag = por_49_fired
			country_event = { id = por.49 hours = 6 }
		}
		if = {
			limit = { date > 2029.6.14 date < 2029.7.15 NOT = { has_country_flag = por_50_fired } }
			set_country_flag = por_50_fired
			country_event = { id = por.50 hours = 6 }
		}
		if = {
			limit = { date > 2029.10.4 date < 2029.11.5 NOT = { has_country_flag = por_51_fired } }
			set_country_flag = por_51_fired
			country_event = { id = por.51 hours = 6 }
		}
		# The hint for the hidden phone call. It needs the same narrative
		# prerequisite as the call itself (the Council of State audience).
		if = {
			limit = { date > 2029.11.19 date < 2029.12.20 has_country_flag = POR_ronaldo_legitimacy NOT = { has_country_flag = por_68_fired } }
			set_country_flag = por_68_fired
			country_event = { id = por.68 hours = 20 }
		}
	}
	else_if = {
		limit = { date < 2030.8.1 }
		if = {
			limit = { date < 2030.2.1 has_country_flag = POR_ronaldo_legitimacy NOT = { has_country_flag = por_68_fired } }
			set_country_flag = por_68_fired
			country_event = { id = por.68 hours = 20 }
		}
		# The call window closed unanswered: the opportunity is gone, quietly.
		if = {
			limit = {
				date > 2030.3.16
				has_country_flag = POR_ronaldo_legitimacy
				NOT = { has_country_flag = POR_call_answered }
				NOT = { has_country_flag = POR_call_missed }
			}
			set_country_flag = POR_call_missed
			country_event = { id = por.74 days = 17 }
		}
		if = {
			limit = { date > 2030.4.24 date < 2030.5.15 NOT = { has_country_flag = POR_call_accepted } NOT = { has_country_flag = por_52_fired } }
			set_country_flag = por_52_fired
			country_event = { id = por.52 hours = 6 }
		}
		if = {
			limit = { date > 2030.5.18 date < 2030.6.5 has_country_flag = POR_call_accepted NOT = { has_country_flag = por_71_fired } }
			set_country_flag = por_71_fired
			news_event = { id = por.71 hours = 20 }
		}
		if = {
			limit = { date > 2030.6.9 has_country_flag = por_71_fired NOT = { has_country_flag = por_72_fired } }
			set_country_flag = por_72_fired
			news_event = { id = por.72 hours = 6 }
		}
	}
}
