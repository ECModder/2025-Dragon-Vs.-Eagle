# Baseline shares of every state from its 2024 margin, the national mood and the chosen bloc.
# In the first cycle, until the midterms (USA_phase < 2) every Republican is a Trumpist (bloc 5); in later cycles the GOP is always bloc 4. Afterwards the GOP is the neoconservatives (bloc 4) unless the Trumpist third-mandate ticket is chosen.
USA_race_baseline_effect = {
	for_loop_effect = {
		start = 0
		end = 51
		value = i
		set_temp_variable = { usb_me = USA_mood }
		set_temp_variable = { usb_w = USA_mw^i }
		multiply_temp_variable = { usb_me = usb_w }
		set_temp_variable = { usb_m = USA_m0^i }
		subtract_from_temp_variable = { usb_m = usb_me }
		set_temp_variable = { usb_r = usb_m }
		divide_temp_variable = { usb_r = 2 }
		add_to_temp_variable = { usb_r = 50 }
		clamp_temp_variable = { var = usb_r min = 4 max = 96 }
		set_temp_variable = { usb_d = 100 }
		subtract_from_temp_variable = { usb_d = usb_r }
		set_temp_variable = { usb_so = 0 }
		set_temp_variable = { usb_ld = 0 }
		set_temp_variable = { usb_cd = usb_d }
		set_temp_variable = { usb_gt = usb_r }
		set_temp_variable = { usb_gn = 0 }
		if = {
			limit = { check_variable = { USA_phase > 1 } }
			set_temp_variable = { usb_gt = 0 }
			set_temp_variable = { usb_gn = usb_r }
		}
		if = {
			limit = { check_variable = { USA_cycle > 0 } }
			set_temp_variable = { usb_gt = 0 }
			set_temp_variable = { usb_gn = usb_r }
		}
		if = {
			limit = { check_variable = { USA_chosen = 1 } }
			set_temp_variable = { usb_so = USA_soc^i }
			set_temp_variable = { usb_cd = usb_d }
			subtract_from_temp_variable = { usb_cd = usb_so }
		}
		if = {
			limit = { check_variable = { USA_chosen = 2 } }
			set_temp_variable = { usb_ld = usb_d }
			multiply_temp_variable = { usb_ld = USA_ldf^i }
			set_temp_variable = { usb_cd = usb_d }
			subtract_from_temp_variable = { usb_cd = usb_ld }
		}
		if = {
			limit = {
				check_variable = { USA_chosen = 5 }
				check_variable = { USA_phase > 1 }
				check_variable = { USA_cycle = 0 }
			}
			set_temp_variable = { usb_gt = usb_r }
			multiply_temp_variable = { usb_gt = USA_gtf^i }
			set_temp_variable = { usb_gn = usb_r }
			subtract_from_temp_variable = { usb_gn = usb_gt }
		}
		set_temp_variable = { usb_k = i }
		multiply_temp_variable = { usb_k = 5 }
		set_variable = { USA_bs^usb_k = usb_so }
		add_to_temp_variable = { usb_k = 1 }
		set_variable = { USA_bs^usb_k = usb_ld }
		add_to_temp_variable = { usb_k = 1 }
		set_variable = { USA_bs^usb_k = usb_cd }
		add_to_temp_variable = { usb_k = 1 }
		set_variable = { USA_bs^usb_k = usb_gn }
		add_to_temp_variable = { usb_k = 1 }
		set_variable = { USA_bs^usb_k = usb_gt }
	}
}

# Keeps a copy of the old baseline, recomputes it, and moves the shares by the difference (mood drift).

	for_loop_effect = {
		start = 0
		end = 255
		value = k
		set_temp_variable = { usd_v = USA_bs^k }
		set_variable = { USA_bo^k = usd_v }
	}
	USA_race_baseline_effect = yes
	for_loop_effect = {
		start = 0
		end = 255
		value = k
		set_temp_variable = { usd_n = USA_bs^k }
		set_temp_variable = { usd_o = USA_bo^k }
		subtract_from_temp_variable = { usd_n = usd_o }
		add_to_variable = { USA_sh^k = usd_n }
		clamp_variable = { var = USA_sh^k min = 0 }
	}
}

# Gains fade: every month the shares move 10% of the way back to the baseline.

		end = 255

		set_temp_variable = { usf_s = USA_sh^k }

	}
}

# Winners, electoral votes, the EV bar, the balance of power and the selected-state card.
# A state goes to the camp (Democratic = blocs 1-3, Republican = blocs 4-5) whose shares add up to more; it takes the colour of the strongest bloc inside that camp.
USA_race_recalc_effect = {
	for_loop_effect = {
		start = 1
		end = 6
		value = b
		set_variable = { USA_evb^b = 0 }
		set_variable = { USA_stb^b = 0 }
	}
	for_loop_effect = {
		start = 0
		end = 51
		value = i
		set_temp_variable = { usw_best = -1 }
		set_temp_variable = { usw_bw = 1 }
		set_temp_variable = { usw_ix = i }
		multiply_temp_variable = { usw_ix = 5 }
		set_temp_variable = { usw_k = usw_ix }
		set_temp_variable = { usw_dm = USA_sh^usw_k }
		add_to_temp_variable = { usw_k = 1 }
		add_to_temp_variable = { usw_dm = USA_sh^usw_k }
		add_to_temp_variable = { usw_k = 1 }
		add_to_temp_variable = { usw_dm = USA_sh^usw_k }
		for_loop_effect = {
			start = 0
			end = 5
			value = b
			set_temp_variable = { usw_ok = 0 }
			if = {
				limit = {
					check_variable = { usw_dm > 50 }
					check_variable = { b < 3 }
				}
				set_temp_variable = { usw_ok = 1 }
			}
			if = {
				limit = {
					NOT = { check_variable = { usw_dm > 50 } }
					check_variable = { b > 2 }
				}
				set_temp_variable = { usw_ok = 1 }
			}
			set_temp_variable = { usw_k = usw_ix }
			add_to_temp_variable = { usw_k = b }
			set_temp_variable = { usw_v = USA_sh^usw_k }
			if = {
				limit = {
					check_variable = { usw_ok = 1 }
					check_variable = { usw_v > usw_best }
				}
				set_temp_variable = { usw_best = usw_v }
				set_temp_variable = { usw_bw = b }
				add_to_temp_variable = { usw_bw = 1 }
			}
		}
		set_variable = { USA_win^i = usw_bw }
		set_variable = { USA_fr^i = usw_bw }
		set_temp_variable = { usw_ev = USA_ev^i }
		add_to_variable = { USA_evb^usw_bw = usw_ev }
		add_to_variable = { USA_stb^usw_bw = 1 }
	}
	set_variable = { USA_ev_soc = USA_evb^1 }
	set_variable = { USA_ev_ld = USA_evb^2 }
	set_variable = { USA_ev_cd = USA_evb^3 }
	set_variable = { USA_ev_gn = USA_evb^4 }
	set_variable = { USA_ev_gt = USA_evb^5 }
	set_variable = { USA_st_soc = USA_stb^1 }
	set_variable = { USA_st_ld = USA_stb^2 }
	set_variable = { USA_st_cd = USA_stb^3 }
	set_variable = { USA_st_gn = USA_stb^4 }
	set_variable = { USA_st_gt = USA_stb^5 }
	set_variable = { USA_dem_ev = USA_ev_soc }
	add_to_variable = { USA_dem_ev = USA_ev_ld }
	add_to_variable = { USA_dem_ev = USA_ev_cd }
	set_variable = { USA_gop_ev = USA_ev_gn }
	add_to_variable = { USA_gop_ev = USA_ev_gt }
	# EV bar: 54 cells of 10 votes, coloured by cumulative votes (left to right: socialist, left dems, centrist dems, neocons, trumpists)
	set_temp_variable = { usc_1 = USA_ev_soc }
	set_temp_variable = { usc_2 = usc_1 }
	add_to_temp_variable = { usc_2 = USA_ev_ld }
	set_temp_variable = { usc_3 = usc_2 }
	add_to_temp_variable = { usc_3 = USA_ev_cd }
	set_temp_variable = { usc_4 = usc_3 }
	add_to_temp_variable = { usc_4 = USA_ev_gn }
	for_loop_effect = {
		start = 0
		end = 54
		value = j
		set_temp_variable = { usc_mid = j }
		multiply_temp_variable = { usc_mid = 10 }
		add_to_temp_variable = { usc_mid = 5 }
		set_temp_variable = { usc_f = 5 }
		if = {
			limit = { check_variable = { usc_mid < usc_4 } }
			set_temp_variable = { usc_f = 4 }
		}
		if = {
			limit = { check_variable = { usc_mid < usc_3 } }
			set_temp_variable = { usc_f = 3 }
		}
		if = {
			limit = { check_variable = { usc_mid < usc_2 } }
			set_temp_variable = { usc_f = 2 }
		}
		if = {
			limit = { check_variable = { usc_mid < usc_1 } }
			set_temp_variable = { usc_f = 1 }
		}
		set_variable = { USA_evbar_fr^j = usc_f }
	}
	# highlight of the chosen bloc in the table (frame 1 clear, 2 lit)
	set_variable = { USA_hl1 = 1 }
	set_variable = { USA_hl2 = 1 }
	set_variable = { USA_hl3 = 1 }
	set_variable = { USA_hl4 = 1 }
	set_variable = { USA_hl5 = 1 }
	if = { limit = { check_variable = { USA_chosen = 1 } } set_variable = { USA_hl1 = 2 } }
	if = { limit = { check_variable = { USA_chosen = 2 } } set_variable = { USA_hl2 = 2 } }
	if = { limit = { check_variable = { USA_chosen = 3 } } set_variable = { USA_hl3 = 2 } }
	if = { limit = { check_variable = { USA_chosen = 4 } } set_variable = { USA_hl4 = 2 } }
	if = { limit = { check_variable = { USA_chosen = 5 } } set_variable = { USA_hl5 = 2 } }
	USA_race_bop_effect = yes
	USA_card_refresh_effect = yes
}

# Balance-of-power bar: value = (GOP votes - Democratic votes) / 538, in steps of 0.1.
USA_race_bop_effect = {
	if = {
		limit = { has_country_flag = USA_race_active }
		set_temp_variable = { usp_v = USA_gop_ev }
		subtract_from_temp_variable = { usp_v = USA_dem_ev }
		divide_temp_variable = { usp_v = 538 }
		multiply_temp_variable = { usp_v = 10 }
		round_temp_variable = usp_v

# One campaign action in the selected state (USA_sel) on a theme (USA_theme 1..5) for the chosen bloc (USA_chosen 1..5).
USA_campaign_effect = {
	set_temp_variable = { uca_s = USA_sel }
	set_temp_variable = { uca_t = USA_theme }
	set_temp_variable = { uca_c = USA_chosen }
	set_temp_variable = { uca_p = 6 }
	set_temp_variable = { uca_i1 = USA_iss1^uca_s }
	set_temp_variable = { uca_i2 = USA_iss2^uca_s }
	if = {
		limit = { check_variable = { uca_i1 = uca_t } }
		multiply_temp_variable = { uca_p = 1.5 }
	}
	else_if = {
		limit = { check_variable = { uca_i2 = uca_t } }
		multiply_temp_variable = { uca_p = 1.15 }
	}
	else = {
		multiply_temp_variable = { uca_p = 0.6 }
	}
	set_temp_variable = { uca_f = uca_c }
	subtract_from_temp_variable = { uca_f = 1 }
	multiply_temp_variable = { uca_f = 5 }
	add_to_temp_variable = { uca_f = uca_t }
	subtract_from_temp_variable = { uca_f = 1 }
	multiply_temp_variable = { uca_p = USA_fit^uca_f }
	multiply_temp_variable = { uca_p = USA_hcap^uca_c }
	set_temp_variable = { uca_e = USA_el^uca_s }
	if = {
		limit = {
			OR = {
				check_variable = { uca_c = 1 }
				check_variable = { uca_c = 2 }
				check_variable = { uca_c = 5 }
			}
		}
		clamp_temp_variable = { var = uca_e min = 0.85 }
	}
	multiply_temp_variable = { uca_p = uca_e }
	# Strength = 6 points x issue match x bloc credibility on the theme x bloc handicap x state elasticity (minor blocs ignore the low elasticity of safe states)
	# x distance factor: the further the state sits from the bloc on the left-right axis, the harder it is
	set_temp_variable = { uca_ix = uca_s }
	multiply_temp_variable = { uca_ix = 5 }
	set_temp_variable = { uca_cg = 0 }
	for_loop_effect = {
		start = 0
		end = 5
		value = b
		set_temp_variable = { uca_k = uca_ix }
		add_to_temp_variable = { uca_k = b }
		set_temp_variable = { uca_v = USA_sh^uca_k }
		set_temp_variable = { uca_pos = b }
		add_to_temp_variable = { uca_pos = 1 }
		multiply_temp_variable = { uca_v = uca_pos }
		add_to_temp_variable = { uca_cg = uca_v }
	}
	divide_temp_variable = { uca_cg = 100 }
	set_temp_variable = { uca_d = uca_c }
	subtract_from_temp_variable = { uca_d = uca_cg }
	if = {
		limit = { check_variable = { uca_d < 0 } }
		multiply_temp_variable = { uca_d = -1 }
	}
	multiply_temp_variable = { uca_d = 0.2 }
	set_temp_variable = { uca_m = 1 }
	subtract_from_temp_variable = { uca_m = uca_d }
	clamp_temp_variable = { var = uca_m min = 0.25 max = 1 }
	multiply_temp_variable = { uca_p = uca_m }
	# apply: the chosen bloc gains, the others lose proportionally
	set_temp_variable = { uca_kc = uca_ix }
	add_to_temp_variable = { uca_kc = uca_c }
	subtract_from_temp_variable = { uca_kc = 1 }
	set_temp_variable = { uca_old = USA_sh^uca_kc }
	set_temp_variable = { uca_new = uca_old }
	add_to_temp_variable = { uca_new = uca_p }
	clamp_temp_variable = { var = uca_new max = 92 }
	if = {
		limit = { check_variable = { uca_new > uca_old } }
		set_temp_variable = { uca_num = 100 }
		subtract_from_temp_variable = { uca_num = uca_new }
		set_temp_variable = { uca_den = 100 }
		subtract_from_temp_variable = { uca_den = uca_old }
		divide_temp_variable = { uca_num = uca_den }
		set_temp_variable = { uca_cb = uca_c }
		subtract_from_temp_variable = { uca_cb = 1 }
		for_loop_effect = {
			start = 0
			end = 5
			value = b
			set_temp_variable = { uca_k = uca_ix }
			add_to_temp_variable = { uca_k = b }
			if = {
				limit = { check_variable = { b = uca_cb } }
				set_variable = { USA_sh^uca_k = uca_new }
			}
			else = {
				multiply_variable = { USA_sh^uca_k = uca_num }
			}
		}
	}
	set_variable = { USA_cool^uca_s = 6 }
	add_political_power = -25
	USA_race_recalc_effect = yes
}

			limit = { check_variable = { upw_camp = 1 } }

# House and Senate seats from the Democratic share of every state (called after the midterms and after the presidential election).
USA_congress_seats_effect = {
		set_variable = { USA_house_d = 0 }
		set_variable = { USA_senate_d = 0 }
		for_loop_effect = {
			start = 0
			end = 51
			value = i
			set_temp_variable = { usm_ev = USA_ev^i }
			set_temp_variable = { usm_ix = i }
			multiply_temp_variable = { usm_ix = 5 }
			set_temp_variable = { usm_k = usm_ix }
			set_temp_variable = { usm_dm = USA_sh^usm_k }
			add_to_temp_variable = { usm_k = 1 }
			add_to_temp_variable = { usm_dm = USA_sh^usm_k }
			add_to_temp_variable = { usm_k = 1 }
			add_to_temp_variable = { usm_dm = USA_sh^usm_k }
			set_temp_variable = { usm_hs = usm_ev }
			subtract_from_temp_variable = { usm_hs = 2 }
			multiply_temp_variable = { usm_hs = usm_dm }
			divide_temp_variable = { usm_hs = 100 }
			add_to_variable = { USA_house_d = usm_hs }
			if = {
				limit = {
					NOT = { check_variable = { i = 8 } }
					check_variable = { usm_dm > 50 }
				}
				add_to_variable = { USA_senate_d = 2 }
			}
		}
		round_variable = USA_house_d
		set_variable = { USA_house_gop = 435 }
		subtract_from_variable = { USA_house_gop = USA_house_d }
		set_variable = { USA_senate_gop = 100 }
		subtract_from_variable = { USA_senate_gop = USA_senate_d }
		set_variable = { global.USA_res_house_d = USA_house_d }
		set_variable = { global.USA_res_house_r = USA_house_gop }
		set_variable = { global.USA_res_senate_d = USA_senate_d }
		set_variable = { global.USA_res_senate_r = USA_senate_gop }
		set_variable = { global.USA_res_dem_ev = USA_dem_ev }
		set_variable = { global.USA_res_gop_ev = USA_gop_ev }
		set_variable = { representatives_support = USA_house_gop }
		set_variable = { senators_support = USA_senate_gop }
}
