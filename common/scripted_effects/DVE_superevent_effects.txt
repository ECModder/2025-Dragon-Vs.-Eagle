# ============================================================
#   DvE - SUPER EVENTS
#   A super event is a number: 1 INT, 2 SRB, 3 SOV, 4 RCA win
#   their civil war, 5 the 2nd Romanian Revolution. Each fires
#   once, for every human player, each with its own song. The
#   song is a sound effect (sound/dve_superevents) started by the
#   window DVE_superevent_sound_<n>_window when the super event
#   shows up, so it plays even with the music playlist stopped.
#   To add one: a frame in superevent_images.dds (and noOfFrames
#   in DVE_superevent.gfx), the five DVE_SE_<n>_* loc keys, a
#   block in each of the five defined_text, a wrapper like the
#   ones below, and a sound + window + scripted_gui for its song
#   (dve_superevents_sounds.asset, DVE_superevent_sounds.gui,
#   DVE_superevent_sounds_gui.txt).
# ============================================================

# Shows super event number dve_superevent_id (temp variable).
DVE_superevent_fire_effect = {
	every_country = {
		limit = { is_ai = no }
		set_variable = { DVE_superevent = dve_superevent_id }
	}
}

# INT wins its civil war.
DVE_superevent_INT_victory_effect = {
	if = {
		limit = { NOT = { has_global_flag = DVE_superevent_INT_victory } }
		set_global_flag = DVE_superevent_INT_victory
		set_temp_variable = { dve_superevent_id = 1 }
		DVE_superevent_fire_effect = yes
	}
}
