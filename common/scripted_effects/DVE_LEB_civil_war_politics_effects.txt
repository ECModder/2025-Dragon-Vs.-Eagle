# Scope: LEB. Run once when the war starts (from LEB_civil_war_effect).
# General Joseph Aoun heads the emergency administration: reactionary group,
# sub-ideology military_democracy (martial law, institutions kept, no elections
# until the war ends). Hariri (reactionary) and Geagea (conservative) stay in
# the LEB history as fallback party leaders, so LEB is never leaderless.
LEB_cw_adwan_takes_over_effect = {
	if = {
		limit = { NOT = { has_character = LEB_Aoun } }
		recruit_character = LEB_Aoun
	}
	# Wartime picture (beret), registered in interface/DVE_LEB_leaders.gfx.
	LEB_Aoun = {
		set_portraits = { civilian = { large = GFX_portrait_leb_aoun_general } }
	}
	set_politics = {
		ruling_party = reactionary
		elections_allowed = no
	}
	set_popularities = {
		totalitarian_socialism = 0
		authoritarian_socialism = 0
		vanguard_socialism = 1
		revolutionary_socialism = 1
		progressive = 8
		liberal = 7
		centrist = 6
		conservative = 20
		reactionary = 41
		authoritarianism = 3
		autocracy = 4
		absolutism = 2
		ultranationalism = 5
		fundamentalism = 2
	}
	set_party_name = { ideology = fundamentalism name = "[Banned] Hezbollah" long_name = "[Banned] Hezbollah" }

	# After the ruling party is set: Aoun (second role of LEB_Aoun) = country leader.
	promote_character = { character = LEB_Aoun ideology = military_democracy }

	set_country_flag = LEB_emergency_administration
	set_country_flag = LEB_aoun_in_charge
}

# Same effect under the name of the first draft.
LEB_cw_government_effect = {
	LEB_cw_adwan_takes_over_effect = yes
}

# Scope: LEB. Beirut goes to the exclusion zone: the capital moves.
# Call after BEZ has taken state 553.
LEB_cw_capital_effect = {
	if = {
		limit = {
			NOT = { owns_state = 553 }
			owns_state = 1074
		}
		set_capital = { state = 1074 } # Mount Lebanon
	}
}

# Scope: LEB. The Sunni and Druze factions join the government
# ("The Emergency Coalition"). LEB already holds Hariri (reactionary)
# and Jumblat (authoritarianism) as party leaders from its history.
LEB_cw_emergency_coalition_politics_effect = {
	if = {
		limit = { NOT = { has_country_flag = LEB_cw_coalition_politics_done } }
		set_country_flag = LEB_cw_coalition_politics_done
		add_popularity = { ideology = reactionary popularity = 0.07 }
		add_popularity = { ideology = authoritarianism popularity = 0.05 }
	}
}
