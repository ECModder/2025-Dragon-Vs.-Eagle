# ============================================================
#   DvE - L'INTERNAZIONALE SOVIETICA: gli effetti condivisi
#   Girano tutti NELLO SCOPE DEL BERSAGLIO, con ROOT = SOV.
# ============================================================

# ------------------------------------------------------------
#   L'influenza: spinge il bersaglio verso l'ideologia che
#   governa a Mosca. add_popularity vuole un'ideologia scritta
#   per esteso, non si puo' passare quella di ROOT: da qui la
#   catena di if, una per ciascuna delle tre internazionali.
# ------------------------------------------------------------
SOV_intl_influence_effect = {

	if = {
		limit = { ROOT = { has_government = totalitarian_socialism } }
		add_popularity = { ideology = totalitarian_socialism popularity = 0.07 }
	}
	else_if = {
		limit = { ROOT = { has_government = vanguard_socialism } }
		add_popularity = { ideology = vanguard_socialism popularity = 0.07 }
	}
	else_if = {
		limit = { ROOT = { has_government = revolutionary_socialism } }
		add_popularity = { ideology = revolutionary_socialism popularity = 0.07 }
	}

	# L'agitazione lascia il segno anche quando non converte.
	add_stability = -0.02
}


# ------------------------------------------------------------
#   L'annessione delle ex repubbliche e dei satelliti.
#   Marca gli stati acquisiti con una flag che porta il tag di
#   provenienza (SOV_intl_land_@FROM diventa p.es.
#   SOV_intl_land_UKR): serve alla decisione dei core, che
#   altrimenti non saprebbe piu' quali terre venivano da chi.
# ------------------------------------------------------------
SOV_intl_annex_effect = {

	every_owned_state = {
		set_state_flag = SOV_intl_land_@PREV
	}

	ROOT = {
		set_country_flag = SOV_intl_annexed_@PREV
		annex_country = { target = PREV transfer_troops = yes }
	}
}
