# ============================================================
#   DvE - LE RIVOLUZIONI ROSSE DELL'ALBERO ITALIANO
#
#   Usati dai focus LT_redfra, LT_redspa, LT_redger, LT_redeng
#   (common/national_focus/italy.txt).
#
#   PERCHE' NON start_civil_war
#   start_civil_war accetta ideology/size/states/capital ma NON un
#   tag: creerebbe nazioni dinamiche (D01, D02...) invece di FSR,
#   SPA, GDR e SRB. Come per la guerra civile russa
#   (DVE_RUS_effects.txt) i tag vengono attivati davvero: ricevono
#   territorio e cores, spostano la capitale e dichiarano guerra.
#
#   Il trasferimento passa sempre da every_owned_state con un
#   limit: cosi' si spostano SOLO gli stati che il paese possiede
#   ancora al momento del focus. Un 'transfer_state' diretto li
#   prenderebbe a chiunque li possieda, rubandoli a terzi se nel
#   frattempo il paese e' stato invaso.
#
#   Governo, popolarita' e stabilita' dei tag socialisti NON
#   vengono forzati qui: restano quelli dei rispettivi file in
#   history/countries/, cosi' basta modificare quelli.
# ============================================================


# ============================================================
#   FRANCIA -> FSR (LT_redfra)
#   La rivoluzione prende Parigi, il nord industriale, l'est e la
#   valle del Rodano. Il governo si rifugia in Aquitania.
#   20 stati su 26 metropolitani (~77%, era size = 0.75).
# ============================================================
ITA_red_france_effect = {

	FRA = {
		every_owned_state = {
			limit = {
				OR = {
					state = 1     # France
					state = 14    # Brittany
					state = 15    # Normandy
					state = 16    # Ile de France - Parigi
					state = 17    # Alsace-Lorraine
					state = 18    # Champagne
					state = 20    # Franche-Comte
					state = 21    # Bouches-du-Rhone
					state = 24    # Centre
					state = 25    # Limousin
					state = 26    # Bourgogne
					state = 27    # Champagne 2
					state = 28    # Alsace
					state = 29    # Pas de Calais
					state = 30    # Loire
					state = 32    # Rhone-Alpes
					state = 33    # Centre sud
					state = 735   # Savoy
					state = 785   # Picardy
					state = 851   # Var
				}
			}
			add_core_of = FSR
			FSR = { transfer_state = PREV }
		}

		# quel che resta del governo si sposta a Bordeaux
		if = {
			limit = {
				NOT = { owns_state = 16 }
				owns_state = 19
			}
			set_capital = { state = 19 }
		}
	}


	# --- la Guardia Rossa ---
	# Il nucleo armato della rivoluzione: non e' l'esercito
	# regolare, che resta col governo, ma quello che i comitati
	# riescono a mettere in piedi nelle prime settimane.
	if = {
		limit = { country_exists = FSR }
		FSR = {
			hidden_effect = {
				if = {
					limit = { NOT = { has_template = "Garde Rouge" } }
					division_template = {
						name = "Garde Rouge"
						regiments = {
							infantry = { x = 0 y = 0 }  infantry = { x = 0 y = 1 }  infantry = { x = 0 y = 2 }
							infantry = { x = 1 y = 0 }  infantry = { x = 1 y = 1 }  infantry = { x = 1 y = 2 }
						}
						support = {
							engineer = { x = 0 y = 0 }
						}
					}
				}
			}
			capital_scope = {
				create_unit = {
					division = "name = \"Garde Rouge\" division_template = \"Garde Rouge\" start_experience_factor = 0.25 start_equipment_factor = 0.60"
					owner = FSR
					count = 14
					allow_spawning_on_enemy_provs = yes
				}
			}
		}
	}

	if = {
		limit = { country_exists = FSR }
		FSR = {
			if = {
				limit = { owns_state = 16 }
				set_capital = { state = 16 }
			}
			declare_war_on = { target = FRA type = annex_everything }
		}
	}
}
