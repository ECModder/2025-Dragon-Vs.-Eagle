# ============================================================
#   DvE - LA GUERRA D'UCRAINA
#
#   Il gioco parte con RUS e UKR gia' in guerra (on_startup, in
#   common/on_actions/DVE_on_actions.txt).
#
#   MECCANICA, sul modello della guerra italo-etiope del gioco
#   base: una missione a tempo con una scala di obiettivi.
#
#   RUSSIA - 278 giorni per, in quest'ordine:
#     1) tenere le quattro regioni del Donbass  (Kherson,
#        Zaporozhe, Donetsk, Lugansk)
#     2) prendere Odessa
#     3) prendere Kiev
#
#   UCRAINA - gli stessi 278 giorni per, in quest'ordine:
#     1) riprendere le quattro regioni del Donbass
#     2) riprendere la Crimea
#     3) portare la guerra in Russia (Rostov o Belgorod)
#
#   I TRE ESITI
#     A) Kiev cade                -> l'Ucraina diventa un puppet
#        russo e Mosca tiene tutto cio' che occupa
#     B) il Donbass torna ucraino -> "Lost in Ukraine": trattato,
#        albero post-sconfitta, crisi di successione
#     C) scadono i 278 giorni     -> pace bianca
#
#   Il controllo degli obiettivi non gira a intervalli: si aggancia
#   a on_state_control_changed, che scatta esattamente quando uno
#   stato cambia mano. Ogni esito e' protetto da una flag, quindi
#   non puo' avvenire due volte.
# ============================================================

# --- gli stati che contano ---
#   Donbass  196 Kherson | 200 Zaporozhe | 227 Donetsk | 228 Lugansk
#   Crimea   137
#   Ucraina  192 Odessa  | 202 Kiev
#   Russia   218 Rostov  | 240 Belgorod


# ============================================================
#   1) L'INIZIO - chiamato da on_startup in scope RUS
# ============================================================
DVE_ukraine_war_start_effect = {

	set_country_flag = RUS_ukraine_war_active
	UKR = { set_country_flag = UKR_ukraine_war_active }

	declare_war_on = { target = UKR type = annex_everything }

	# Il Donbass Mosca lo tiene gia' a inizio partita: il primo
	# obiettivo e' la posizione di partenza, non una conquista.
	# Va segnato subito, in silenzio, altrimenti Odessa resta
	# bloccata per sempre - il controllo degli obiettivi gira su
	# on_state_control_changed, e se nessuno stato cambia mano non
	# scatta mai.
	if = {
		limit = {
			controls_state = 196
			controls_state = 200
			controls_state = 227
			controls_state = 228
		}
		set_country_flag = RUS_obj_donbass
	}
}


# ============================================================
#   2) IL CONTROLLO DEGLI OBIETTIVI
#      Chiamato in scope RUS da on_state_control_changed.
# ============================================================
DVE_ukraine_war_check_effect = {

	if = {
		limit = { has_country_flag = RUS_ukraine_war_active
		          NOT = { has_country_flag = RUS_ukraine_war_over } }

		# ---------- ESITO B: il Donbass torna ucraino ----------
		# Va controllato per primo: se il Donbass e' perduto la
		# guerra e' finita, qualunque cosa Mosca stia occupando.
		if = {
			limit = {
				UKR = {
					controls_state = 196
					controls_state = 200
					controls_state = 227
					controls_state = 228
				}
			}
			RUS_ukraine_defeat_effect = yes
		}
		else = {
			# ---------- scala russa ----------
			if = {
				limit = {
					NOT = { has_country_flag = RUS_obj_donbass }
					controls_state = 196
					controls_state = 200
					controls_state = 227
					controls_state = 228
				}
				set_country_flag = RUS_obj_donbass
				country_event = { id = dve_ukrwar.1 hours = 6 }
			}
			if = {
				limit = {
					has_country_flag = RUS_obj_donbass
					NOT = { has_country_flag = RUS_obj_odessa }
					controls_state = 192
				}
				set_country_flag = RUS_obj_odessa
				country_event = { id = dve_ukrwar.2 hours = 6 }
			}
			# ---------- ESITO A: Kiev ----------
			if = {
				limit = {
					has_country_flag = RUS_obj_odessa
					controls_state = 202
				}
				RUS_ukraine_victory_effect = yes
			}

			# ---------- scala ucraina ----------
			UKR = {
				if = {
					limit = {
						NOT = { has_country_flag = UKR_obj_crimea }
						controls_state = 137
					}
					set_country_flag = UKR_obj_crimea
					country_event = { id = dve_ukrwar.11 hours = 6 }
				}
				if = {
					limit = {
						NOT = { has_country_flag = UKR_obj_into_russia }
						OR = { controls_state = 218 controls_state = 240 }
					}
					set_country_flag = UKR_obj_into_russia
					country_event = { id = dve_ukrwar.12 hours = 6 }
				}
			}
		}
	}
}


# ============================================================
#   3) ESITO A - Kiev cade
# ============================================================
RUS_ukraine_victory_effect = {

	set_country_flag = RUS_ukraine_war_over
	set_country_flag = RUS_won_the_war_in_ukraine
	UKR = { set_country_flag = UKR_ukraine_war_over }

	# Mosca si tiene tutto cio' che le sue truppe occupano
	UKR = {
		every_owned_state = {
			limit = { is_controlled_by = ROOT }
			ROOT = { transfer_state = PREV }
		}
	}

	white_peace = UKR
	puppet = UKR

	add_stability = 0.10
	add_war_support = 0.10
	add_political_power = 200

	country_event = { id = dve_ukrwar.3 hours = 2 }
}


# ============================================================
#   4) ESITO B - il Donbass torna ucraino: "Lost in Ukraine"
#      Fa esattamente cio' che fa la decisione di debug
#      DVE_debug_lost_ukraine.
# ============================================================
RUS_ukraine_defeat_effect = {

	set_country_flag = RUS_ukraine_war_over
	UKR = { set_country_flag = UKR_ukraine_war_over }
	UKR = { set_country_flag = UKR_obj_donbass }

	white_peace = UKR

	# il trattato prima, l'albero dopo: e' l'ordine che tiene
	# anche la decisione di debug
	RUS_lost_war_settlement_effect = yes
	load_focus_tree = { tree = russian_defeated_focus keep_completed = no }
	set_country_flag = RUS_lost_the_war_in_ukraine
	add_political_power = 200

	country_event = { id = dve_ukrwar.5 hours = 2 }
	UKR = { country_event = { id = dve_ukrwar.10 hours = 4 } }
}


# ============================================================
#   5) ESITO C - scadono i 278 giorni: pace bianca
#      Protetto da flag: la chiamano entrambe le missioni, ma
#      succede una volta sola.
# ============================================================
DVE_ukraine_white_peace_effect = {

	if = {
		limit = { NOT = { has_country_flag = RUS_ukraine_war_over } }

		set_country_flag = RUS_ukraine_war_over
		set_country_flag = RUS_ukraine_white_peace
		UKR = { set_country_flag = UKR_ukraine_war_over }

		white_peace = UKR

		add_stability = -0.05
		add_war_support = -0.10

		country_event = { id = dve_ukrwar.4 hours = 2 }
		UKR = { country_event = { id = dve_ukrwar.13 hours = 4 } }
	}
}
