# ============================================================
#   DvE - GUERRA CIVILE CINESE
#
#   ATTENZIONE - SPARTIZIONE PROVVISORIA
#   Dei 76 stati della Cina, quelli da 601 a 622 si chiamano
#   "China 1" ... "China 19": dal nome non si capisce dove siano,
#   quindi NON sono assegnati e restano a PRC (il "Centralist
#   Faction" della cartina).
#   Qui sono spartiti solo i 29 stati identificabili per nome.
#   Quando arrivano i confini esatti basta allungare le quattro
#   liste qui sotto: il resto del file non cambia.
#
#   Come per la guerra civile russa non si usa start_civil_war,
#   che creerebbe tag dinamici (D01, D02...) invece di CSR, CCR,
#   PXA e PTA.
#
#   Il trasferimento passa da every_owned_state con un limit, cosi'
#   si spostano solo gli stati che la Cina possiede davvero al
#   momento in cui scatta l'effetto.
#
#   Governo, popolarita' e stabilita' dei quattro tag NON vengono
#   forzati qui: restano quelli dei file in history/countries/.
# ============================================================


# ============================================================
#   CCR - i maoisti, nord del paese
#   Pechino, Hebei, Shanxi, l'Ordos e Yan'an, la citta' santa del maoismo.
#   8 stati.
# ============================================================
CHN_spawn_CCR_effect = {

	every_owned_state = {
		limit = {
			OR = {
				state = 608  # Beijing
				state = 615  # Shanxi
				state = 744  # Xian
				state = 746  # Ordos
				state = 1032 # Yanan
				state = 1039 # Hebei-Chahar
				state = 1045 # Guyuan
				state = 1046 # Yulin
			}
		}
		add_core_of = CCR
		CCR = { transfer_state = PREV }
	}

	if = {
		limit = { country_exists = CCR }
		CCR = {
			if = {
				limit = { owns_state = 608 }
				set_capital = { state = 608 }
			}
		}
	}
}

# ============================================================
#   LA GUERRA CIVILE COMPLETA
# ============================================================
CHN_civil_war_effect = {

	set_country_flag = CHN_civil_war
	set_country_flag = DVE_civil_war_active

	add_stability = -0.30
	add_war_support = 0.10

	CHN_spawn_CCR_effect = yes
	CHN_spawn_CSR_effect = yes
	CHN_spawn_PXA_effect = yes
	CHN_spawn_PTA_effect = yes

	# Non esistono file OOB cinesi in history/units/, quindi i
	# quattro tag creano il proprio template al volo e schierano
	# le divisioni sulla capitale.
	CHN_civil_war_army_effect = yes

	# --- tutti contro tutti ---
	if = { limit = { country_exists = CCR } declare_war_on = { target = CCR type = annex_everything } }
	if = { limit = { country_exists = CSR } declare_war_on = { target = CSR type = annex_everything } }
	if = { limit = { country_exists = PXA } declare_war_on = { target = PXA type = annex_everything } }
	if = { limit = { country_exists = PTA } declare_war_on = { target = PTA type = annex_everything } }

	# i due partiti rivali si combattono anche fra loro
	if = {
		limit = { country_exists = CCR country_exists = CSR }
		CCR = { declare_war_on = { target = CSR type = annex_everything } }
	}
}

# ============================================================
#   ANNULLA TUTTO
#   La Cina si riprende i quattro tag. Serve solo al debug, per
#   rifare la prova da capo.
#
#   ROOT = la Cina (chi lancia l'effetto), PREV = lo stato in
#   corso. Non si puo' usare PREV al posto di ROOT: dentro
#   CCR = { ... } PREV punta a CCR stessa e gli stati tornerebbero
#   a chi li ha appena persi.
# ============================================================
CHN_civil_war_reset_effect = {

	clr_country_flag = CHN_civil_war
	clr_country_flag = DVE_civil_war_active

	if = { limit = { country_exists = CCR } CCR = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = CSR } CSR = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = PXA } PXA = { every_owned_state = { ROOT = { transfer_state = PREV } } } }
	if = { limit = { country_exists = PTA } PTA = { every_owned_state = { ROOT = { transfer_state = PREV } } } }

	every_owned_state = {
		remove_core_of = CCR
		remove_core_of = CSR
		remove_core_of = PXA
		remove_core_of = PTA
	}
}
