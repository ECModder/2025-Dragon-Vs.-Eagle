# Regime adoption for the three paths of romania_uprise.
# The ideology groups are the mod's own (common/ideologies/00_ideologies.txt).
ROM_adopt_absolutist = {
	add_popularity = { ideology = absolutism popularity = 0.30 }
	set_politics = { ruling_party = absolutism elections_allowed = no }
	set_country_flag = ROM_ideology_absolutist
}

ROM_adopt_autocratic = {

	set_politics = { ruling_party = autocracy elections_allowed = no }
	set_country_flag = ROM_ideology_autocratic
}

ROM_adopt_fundamentalist = {

# Kept for compatibility with older content; the redesigned trees name their states explicitly.
ROM_claim_moldova = {
	if = {
		limit = { country_exists = MOL }
		MOL = { every_owned_state = { add_claim_by = ROM } }
	}
}

ROM_claim_bukovina = {
	if = {
		limit = { country_exists = UKR }
		UKR = {
			every_owned_state = {
				limit = { any_neighbor_state = { is_owned_by = ROM } }
				add_claim_by = ROM
			}
		}
	}
}
