# ============================================================
#   DvE - SECOND LEBANESE CIVIL WAR
#   Five factions, every one against every other:
#     LEB  Lebanese Government: Mount Lebanon (1074), Bekaa (1073)
#     HSC  Hariri Salvation Council: Tripoli-Akkar (1075)
#     HZB  Islamic Resistance in Lebanon: Baalbek-Hermel (1076),
#          Al-Janoub (1113), Aqsa Al-Janoub (1072)
#     BEZ  Beirut Exclusion Zone: Beirut (553)
#     DUZ  Druze Mountain Authority: Chouf-Aley (1112)
#   No date: the future Lebanese focus tree (or the debug decision)
#   calls LEB_civil_war_start_effect, which fires dve_lebcw.1 in LEB
#   scope. Politics and leaders: DVE_LEB_civil_war_politics_effects.txt.
#   Every effect below runs in LEB scope with ROOT = LEB, except
#   the army effects, which run in the scope of their own tag.
# ============================================================

# Starts the war from anywhere (focus, event, debug).
LEB_civil_war_start_effect = {
	if = {
		limit = {
			country_exists = LEB
			NOT = { has_global_flag = LEB_civil_war }
		}
		LEB = { news_event = { id = dve_lebcw.1 } }
	}
}

# --- Spawn effects: states and cores, kit, leader, capital, army ---

LEB_cw_spawn_HSC_effect = {
	every_owned_state = {
		limit = {
			OR = {
				state = 1075 # Tripoli-Akkar
			}
		}
		add_core_of = HSC
		HSC = { transfer_state = PREV }
	}
	if = {
		limit = { country_exists = HSC }
		HSC = {
			DVE_cw_new_faction_setup_effect = yes
			DVE_cw_ensure_leader_effect = { CHAR = HSC_saad_hariri IDEO = paternalism }

				limit = { owns_state = 1075 }
				set_capital = { state = 1075 }

		}
	}
}

# Runs in LEB's scope from dve_lebcw.1 (ROOT = LEB).
LEB_civil_war_effect = {
	set_global_flag = LEB_civil_war
	add_stability = -0.20
	add_war_support = 0.10

	# territory, kit, leader and army of the four breakaway factions
	LEB_cw_spawn_HSC_effect = yes
	LEB_cw_spawn_HZB_effect = yes
	LEB_cw_spawn_BEZ_effect = yes
	LEB_cw_spawn_DUZ_effect = yes

	# Beirut is gone, so the capital moves (politics effects file)
	LEB_cw_capital_effect = yes

	# Hezbollah's national spirits follow Hezbollah
	if = {
		limit = { country_exists = HZB }
		if = {
			limit = { has_idea = LEB_parallel_institutions }
			remove_ideas = LEB_parallel_institutions
			HZB = { add_ideas = LEB_parallel_institutions }
		}
		if = {
			limit = { has_idea = LEB_shadow_of_litani }
			remove_ideas = LEB_shadow_of_litani
			HZB = { add_ideas = LEB_shadow_of_litani }
		}
	}

	# Adwan runs the emergency administration, Aoun replaces him
	# in 30 days (dve_lebcw.3); Geagea stays as fallback leader
	LEB_cw_adwan_takes_over_effect = yes
	country_event = { id = dve_lebcw.3 days = 30 }

	LEB_cw_army_LEB_effect = yes

	# every faction against every other
	every_country = {
		limit = { DVE_cw_lebanon = yes }
		every_country = {
			limit = {
				DVE_cw_lebanon = yes
				NOT = { tag = PREV }
				NOT = { has_war_with = PREV }
				NOT = { is_in_faction_with = PREV }
			}
			PREV = { declare_war_on = { target = PREV type = annex_everything } }
		}
	}

	every_country = {
		limit = { NOT = { DVE_cw_lebanon = yes } }
		news_event = { id = dve_lebcw.2 hours = 6 }
	}
}

# ============================================================
#   THE EMERGENCY COALITION
#   The Sunni and Druze factions join the Government: white
#   peace between LEB, HSC and DUZ, then one faction led by LEB.
#   The wars against HZB and BEZ go on. Called by the future
#   Lebanese focus "The Emergency Coalition" and by the debug
#   decision. Scope: LEB.
# ============================================================
LEB_emergency_coalition_effect = {
	if = {
		limit = {
			has_global_flag = LEB_civil_war
			NOT = { has_global_flag = LEB_emergency_coalition }
			OR = {
				country_exists = HSC
				country_exists = DUZ
			}
		}
		set_global_flag = LEB_emergency_coalition

		# peace first: the wars stop before anybody joins
		if = { limit = { country_exists = HSC } white_peace = HSC }
		if = { limit = { country_exists = DUZ } white_peace = DUZ }
		if = {
			limit = {
				country_exists = HSC
				country_exists = DUZ
			}
			HSC = { white_peace = DUZ }
		}

		# one faction led by LEB
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_faction_leader = yes }
			}
			leave_faction = yes
		}
		if = {
			limit = { is_in_faction = no }
			create_faction = "DVE_FACTION_LEB_UNITY"
		}
		if = {
			limit = { country_exists = HSC }
			HSC = {
				if = {
					limit = {
						is_in_faction = yes
						NOT = { is_in_faction_with = LEB }
					}
					leave_faction = yes
				}
			}
			if = {
				limit = { NOT = { is_in_faction_with = HSC } }
				add_to_faction = HSC
			}
			give_military_access = HSC
			HSC = { give_military_access = LEB }
		}
		if = {
			limit = { country_exists = DUZ }
			DUZ = {
				if = {
					limit = {
						is_in_faction = yes
						NOT = { is_in_faction_with = LEB }
					}
					leave_faction = yes
				}
			}
			if = {
				limit = { NOT = { is_in_faction_with = DUZ } }
				add_to_faction = DUZ
			}
			give_military_access = DUZ
			DUZ = { give_military_access = LEB }
		}

		# Hariri's and Jumblat's followers now count in the Government
		LEB_cw_emergency_coalition_politics_effect = yes
	}
}

# ============================================================
#   RESET (debug only)
#   Puts back what can really be undone: faction, divisions and
#   templates of the five tags, states and cores, capital,
#   politics, leaders, spirits, stability and flags. The
#   equipment already handed out stays in the stockpiles. Scope: LEB.
# ============================================================
LEB_civil_war_reset_effect = {

	# the coalition faction (HSC and DUZ leave with it)
	if = {
		limit = {
			has_global_flag = LEB_emergency_coalition
			is_faction_leader = yes
		}
		dismantle_faction = yes
	}

	# stability and war support of the war start
	if = {
		limit = { has_global_flag = LEB_civil_war }
		add_stability = 0.20
		add_war_support = -0.10
	}

	# politics and leaders of the Government, back to the pre-war values of
	# history/countries/LEB - Lebanon.txt (only if the war changed them)
	if = {
		limit = { has_country_flag = LEB_emergency_administration }
		set_politics = {
			ruling_party = centrist
			elections_allowed = yes
		}
		set_popularities = {
			totalitarian_socialism = 0
			authoritarian_socialism = 0
			vanguard_socialism = 0
			revolutionary_socialism = 0
			progressive = 11
			liberal = 0
			centrist = 22
			conservative = 20
			reactionary = 12
			authoritarianism = 4
			autocracy = 12
			absolutism = 0
			ultranationalism = 1
			fundamentalism = 18
		}
		set_party_name = { ideology = fundamentalism name = "Hezbollah" long_name = "Hezbollah" }
		if = { limit = { has_character = LEB_georges_adwan } retire_character = LEB_georges_adwan }
		if = { limit = { has_character = LEB_Aoun } retire_character = LEB_Aoun }
	}

	# Hezbollah's spirits come back
	if = {
		limit = { NOT = { has_idea = LEB_parallel_institutions } }
		add_ideas = LEB_parallel_institutions
	}
	if = {
		limit = { NOT = { has_idea = LEB_shadow_of_litani } }
		add_ideas = LEB_shadow_of_litani
	}
	if = {
		limit = { country_exists = HZB }
		HZB = {
			if = { limit = { has_idea = LEB_parallel_institutions } remove_ideas = LEB_parallel_institutions }
			if = { limit = { has_idea = LEB_shadow_of_litani } remove_ideas = LEB_shadow_of_litani }
		}
	}

	# divisions, templates and kit flags (while the tags still exist)
	if = {
		limit = { has_template = "Lebanese Army Brigade" }
		delete_unit_template_and_units = { division_template = "Lebanese Army Brigade" disband = yes }
	}
	if = {
		limit = { country_exists = HSC }
		HSC = {
			if = {
				limit = { has_template = "Salvation Council Brigade" }
				delete_unit_template_and_units = { division_template = "Salvation Council Brigade" disband = yes }
			}
			clr_country_flag = DVE_cw_kit_given
		}
	}
	if = {
		limit = { country_exists = HZB }
		HZB = {
			if = {
				limit = { has_template = "Islamic Resistance Brigade" }
				delete_unit_template_and_units = { division_template = "Islamic Resistance Brigade" disband = yes }
			}
			clr_country_flag = DVE_cw_kit_given
			clr_country_flag = LEB_cw_hzb_kit_given
		}
	}
	if = {
		limit = { country_exists = BEZ }
		BEZ = {
			if = {
				limit = { has_template = "Thousand Banners Militia" }
				delete_unit_template_and_units = { division_template = "Thousand Banners Militia" disband = yes }
			}
			clr_country_flag = DVE_cw_kit_given
		}
	}
	if = {
		limit = { country_exists = DUZ }
		DUZ = {
			if = {
				limit = { has_template = "Mountain Guard Brigade" }
				delete_unit_template_and_units = { division_template = "Mountain Guard Brigade" disband = yes }
			}
			clr_country_flag = DVE_cw_kit_given
		}
	}

	# the four factions give back the Lebanese states they hold
	if = {
		limit = { country_exists = HSC }
		HSC = {
			every_owned_state = {
				limit = { OR = { state = 553 state = 1072 state = 1073 state = 1074 state = 1075 state = 1076 state = 1112 state = 1113 } }
				LEB = { transfer_state = PREV }
			}
		}
	}
	if = {
		limit = { country_exists = HZB }
		HZB = {
			every_owned_state = {
				limit = { OR = { state = 553 state = 1072 state = 1073 state = 1074 state = 1075 state = 1076 state = 1112 state = 1113 } }
				LEB = { transfer_state = PREV }
			}
		}
	}
	if = {
		limit = { country_exists = BEZ }
		BEZ = {
			every_owned_state = {
				limit = { OR = { state = 553 state = 1072 state = 1073 state = 1074 state = 1075 state = 1076 state = 1112 state = 1113 } }
				LEB = { transfer_state = PREV }
			}
		}
	}
	if = {
		limit = { country_exists = DUZ }
		DUZ = {
			every_owned_state = {
				limit = { OR = { state = 553 state = 1072 state = 1073 state = 1074 state = 1075 state = 1076 state = 1112 state = 1113 } }
				LEB = { transfer_state = PREV }
			}
		}
	}

	every_owned_state = {
		remove_core_of = HSC
		remove_core_of = HZB
		remove_core_of = BEZ
		remove_core_of = DUZ
	}

	# back to Beirut
	if = {
		limit = { owns_state = 553 }
		set_capital = { state = 553 }
	}

	clr_global_flag = LEB_civil_war
	clr_global_flag = LEB_emergency_coalition
	clr_country_flag = LEB_emergency_administration
	clr_country_flag = LEB_aoun_in_charge
	clr_country_flag = LEB_cw_coalition_politics_done
}
