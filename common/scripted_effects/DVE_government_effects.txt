# Random vote shares for every party, ultranationalists capped at 4%.
# Weights are random^4, not flat random: with 14 flat shares no party
# reaches 20% in ~99% of elections and the vote would repeat forever.
# With ^4 the top party averages ~31% and a re-vote happens ~15% of the time.

	set_temp_variable = { dve_gov_r = random }
	set_temp_variable = { dve_gov_w_totalitarian_socialism = dve_gov_r }

# The formateur takes power with its coalition.
DVE_gov_form_government_effect = {
	if = {
		limit = { has_country_flag = DVE_gov_formateur_totalitarian_socialism }
		set_politics = { ruling_party = totalitarian_socialism elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_authoritarian_socialism }
		set_politics = { ruling_party = authoritarian_socialism elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_progressive }
		set_politics = { ruling_party = progressive elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_liberal }
		set_politics = { ruling_party = liberal elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_centrist }
		set_politics = { ruling_party = centrist elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_conservative }
		set_politics = { ruling_party = conservative elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_reactionary }
		set_politics = { ruling_party = reactionary elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_authoritarianism }
		set_politics = { ruling_party = authoritarianism elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_autocracy }
		set_politics = { ruling_party = autocracy elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_absolutism }
		set_politics = { ruling_party = absolutism elections_allowed = yes }
	}
	else_if = {
		limit = { has_country_flag = DVE_gov_formateur_ultranationalism }
		set_politics = { ruling_party = ultranationalism elections_allowed = yes }
	}
	DVE_gov_end_formation_effect = yes
}
