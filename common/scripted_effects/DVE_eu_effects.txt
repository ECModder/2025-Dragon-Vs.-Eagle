# Calculates national EP delegation by mapping 14 DvE ideologies to 8 EP groups.
DVE_eu_calc_ep_delegation_effect = {
	# Total seats for this country
	set_temp_variable = { DVE_eu_total_seats = DVE_eu_ep_seats }
	# For each DvE ideology, distribute its share to EP groups
	set_temp_variable = { DVE_eu_share = party_popularity@totalitarian_socialism }
	multiply_temp_variable = { DVE_eu_share = 1 }
	multiply_temp_variable = { DVE_eu_share = DVE_eu_total_seats }
	divide_temp_variable = { DVE_eu_share = 100 }
	round_temp_variable = DVE_eu_share
	add_to_variable = { DVE_eu_ep_delegation_ni = DVE_eu_share }
	set_temp_variable = { DVE_eu_share = party_popularity@authoritarian_socialism }
	multiply_temp_variable = { DVE_eu_share = 0.6 }
	multiply_temp_variable = { DVE_eu_share = DVE_eu_total_seats }
	divide_temp_variable = { DVE_eu_share = 100 }
	round_temp_variable = DVE_eu_share
	add_to_variable = { DVE_eu_ep_delegation_sd = DVE_eu_share }
	set_temp_variable = { DVE_eu_share = party_popularity@authoritarian_socialism }
	multiply_temp_variable = { DVE_eu_share = 0.4 }
	multiply_temp_variable = { DVE_eu_share = DVE_eu_total_seats }
	divide_temp_variable = { DVE_eu_share = 100 }
	round_temp_variable = DVE_eu_share
	add_to_variable = { DVE_eu_ep_delegation_left = DVE_eu_share }
	set_temp_variable = { DVE_eu_share = party_popularity@vanguard_socialism }
	multiply_temp_variable = { DVE_eu_share = 0.5 }
	multiply_temp_variable = { DVE_eu_share = DVE_eu_total_seats }
	divide_temp_variable = { DVE_eu_share = 100 }
	round_temp_variable = DVE_eu_share
	add_to_variable = { DVE_eu_ep_delegation_left = DVE_eu_share }
	set_temp_variable = { DVE_eu_share = party_popularity@vanguard_socialism }

# Tallies Council votes: counts states for/against, population for/against.
DVE_eu_council_tally_effect = {
	set_variable = { global.DVE_eu_council_states_for = 0 }
	set_variable = { global.DVE_eu_council_states_against = 0 }
	set_variable = { global.DVE_eu_council_pop_for = 0 }
	set_variable = { global.DVE_eu_council_pop_against = 0 }

	every_country = {
		limit = { has_country_flag = DVE_eu_member }
		if = {
			limit = { check_variable = { DVE_eu_council_vote = 2 } }
			add_to_variable = { global.DVE_eu_council_states_for = 1 }
			add_to_variable = { global.DVE_eu_council_pop_for = DVE_eu_council_weight }
		}
		else_if = {
			limit = { check_variable = { DVE_eu_council_vote = 3 } }
			add_to_variable = { global.DVE_eu_council_states_against = 1 }
			add_to_variable = { global.DVE_eu_council_pop_against = DVE_eu_council_weight }
		}
	}

	# QMV check: 55% states (15) AND 65% population
	set_variable = { global.DVE_eu_council_qmv_pass = 0 }
	if = {
		limit = {
			check_variable = { global.DVE_eu_council_states_for >= 15 }
			check_variable = { global.DVE_eu_council_pop_for >= global.DVE_eu_council_qmv_threshold_pop }
		}
		set_variable = { global.DVE_eu_council_qmv_pass = 1 }
	}
	# Blocking minority check: >=4 states AND >35% population
	set_variable = { global.DVE_eu_council_blocking_minority = 0 }
	if = {
		limit = {
			check_variable = { global.DVE_eu_council_states_against >= 4 }
			check_variable = { global.DVE_eu_council_pop_against > global.DVE_eu_council_blocking_minority_pop }
		}
		set_variable = { global.DVE_eu_council_blocking_minority = 1 }
	}
}

# Parliament vote: simple majority of cast votes (for > against).
DVE_eu_ep_vote_effect = {
	set_variable = { global.DVE_eu_ep_votes_for = 0 }
	set_variable = { global.DVE_eu_ep_votes_against = 0 }
	set_variable = { global.DVE_eu_ep_votes_abstain = 0 }

	every_country = {
		limit = { has_country_flag = DVE_eu_member }
		# Each country's delegation votes as a block (simplified)
		if = {
			limit = { check_variable = { DVE_eu_ep_vote = 2 } }
			add_to_variable = { global.DVE_eu_ep_votes_for = DVE_eu_ep_seats }
		}
		else_if = {
			limit = { check_variable = { DVE_eu_ep_vote = 3 } }
			add_to_variable = { global.DVE_eu_ep_votes_against = DVE_eu_ep_seats }
		}
		else = {
			add_to_variable = { global.DVE_eu_ep_votes_abstain = DVE_eu_ep_seats }
		}
	}

	# EP passes if for > against
	set_variable = { global.DVE_eu_ep_pass = 0 }
	if = {
		limit = { check_variable = { global.DVE_eu_ep_votes_for > global.DVE_eu_ep_votes_against } }
		set_variable = { global.DVE_eu_ep_pass = 1 }
	}
}

		limit = { has_country_flag = DVE_eu_member }
		country_event = { id = dve_eu.40 }
	}
}

# ECI: 1M signatures from 7 countries -> Commission response.
DVE_eu_eci_effect = {
	if = {
		limit = { check_variable = { global.DVE_eu_eci_signatures >= 1000000 } check_variable = { global.DVE_eu_eci_countries >= 7 } }
		country_event = { id = dve_eu.50 }
	}
}

# EDP: deficit >3% GDP -> Council recommendation -> sanctions if persistent.
DVE_eu_edp_check_effect = {
	every_country = {
		limit = { has_country_flag = DVE_eu_eurozone_member }
		if = {
			limit = { check_variable = { budget_deficit_pct_gdp > 3 } }
			if = {
				limit = { NOT = { has_country_flag = DVE_eu_edp_active } }
				set_country_flag = DVE_eu_edp_active
				add_to_array = { global.DVE_eu_edp_countries = THIS }
				country_event = { id = dve_eu.60 }
			}
		}
		else = {
			limit = { has_country_flag = DVE_eu_edp_active }
			clr_country_flag = DVE_eu_edp_active
			remove_from_array = { global.DVE_eu_edp_countries = THIS }
			country_event = { id = dve_eu.61 }
		}
	}
}

# Article 7 TEU: 4/5 Council + EP consent -> sanctions (voting rights suspension).
DVE_eu_article7_effect = {
	if = {
		limit = { check_variable = { global.DVE_eu_article7_stage = 1 } }
		# Council 4/5 majority
		set_variable = { global.DVE_eu_council_article7_for = 0 }
		every_country = {
			limit = { has_country_flag = DVE_eu_member }
			if = { limit = { check_variable = { DVE_eu_council_vote = 2 } } add_to_variable = { global.DVE_eu_council_article7_for = 1 } }
		}
		if = {
			limit = { check_variable = { global.DVE_eu_council_article7_for >= 22 } }
			set_variable = { global.DVE_eu_article7_stage = 2 }
			country_event = { id = dve_eu.70 }
		}
	}
	else_if = {
		limit = { check_variable = { global.DVE_eu_article7_stage = 2 } }
		# EP consent
		DVE_eu_ep_vote_effect = yes
		if = {
			limit = { check_variable = { global.DVE_eu_ep_pass = 1 } }
			set_variable = { global.DVE_eu_article7_stage = 3 }
			country_event = { id = dve_eu.71 }
		}
	}
}
