# Country-level delegation seat calculation based on national party popularities
DVE_eu_country_delegation_calc_effect = {
	# Raw party popularities mapped to ideological shares
	set_temp_variable = { p_tot_soc = party_popularity@totalitarian_socialism }
	set_temp_variable = { p_auth_soc = party_popularity@authoritarian_socialism }
	set_temp_variable = { p_vang_soc = party_popularity@vanguard_socialism }
	set_temp_variable = { p_rev_soc = party_popularity@revolutionary_socialism }
	set_temp_variable = { p_prog = party_popularity@progressive }
	set_temp_variable = { p_lib = party_popularity@liberal }
	set_temp_variable = { p_cent = party_popularity@centrist }
	set_temp_variable = { p_cons = party_popularity@conservative }
	set_temp_variable = { p_react = party_popularity@reactionary }
	set_temp_variable = { p_auth = party_popularity@authoritarianism }
	set_temp_variable = { p_ultra = party_popularity@ultranationalism }
	set_temp_variable = { p_abs = party_popularity@absolutism }
	set_temp_variable = { p_auto = party_popularity@autocracy }
	set_temp_variable = { p_fund = party_popularity@fundamentalism }

	# Group 1: The Left (Frame 2)
	set_variable = { eu_sh_1 = p_rev_soc }
	add_to_variable = { eu_sh_1 = p_vang_soc }
	set_temp_variable = { half_auth_soc = p_auth_soc }
	multiply_variable = { half_auth_soc = 0.5 }
	add_to_variable = { eu_sh_1 = half_auth_soc }

	# Group 2: Greens/EFA (Frame 3)
	set_variable = { eu_sh_2 = p_prog }
	multiply_variable = { eu_sh_2 = 0.4 }

	# Group 3: S&D (Frame 4)
	set_variable = { eu_sh_3 = p_prog }
	multiply_variable = { eu_sh_3 = 0.6 }
	add_to_variable = { eu_sh_3 = half_auth_soc }

	# Group 4: Renew (Frame 5)
	set_variable = { eu_sh_4 = p_lib }
	set_temp_variable = { half_cent = p_cent }
	multiply_variable = { half_cent = 0.4 }
	add_to_variable = { eu_sh_4 = half_cent }

	# Group 5: EPP (Frame 6)
	set_variable = { eu_sh_5 = p_cons }
	multiply_variable = { eu_sh_5 = 0.7 }
	set_temp_variable = { epp_cent = p_cent }
	multiply_variable = { epp_cent = 0.6 }
	add_to_variable = { eu_sh_5 = epp_cent }

	# Group 6: ECR (Frame 7)
	set_variable = { eu_sh_6 = p_cons }
	multiply_variable = { eu_sh_6 = 0.3 }
	set_temp_variable = { ecr_react = p_react }
	multiply_variable = { ecr_react = 0.5 }
	add_to_variable = { eu_sh_6 = ecr_react }
	set_temp_variable = { ecr_auth = p_auth }
	multiply_variable = { ecr_auth = 0.4 }
	add_to_variable = { eu_sh_6 = ecr_auth }

	# Group 7: ID (Frame 8)
	set_variable = { eu_sh_7 = p_react }
	multiply_variable = { eu_sh_7 = 0.5 }
	add_to_variable = { eu_sh_7 = p_ultra }
	set_temp_variable = { id_auth = p_auth }
	multiply_variable = { id_auth = 0.4 }
	add_to_variable = { eu_sh_7 = id_auth }

	# Group 8: NI (Frame 9)
	set_variable = { eu_sh_8 = p_tot_soc }
	add_to_variable = { eu_sh_8 = p_abs }
	add_to_variable = { eu_sh_8 = p_auto }
	add_to_variable = { eu_sh_8 = p_fund }
	set_temp_variable = { ni_auth = p_auth }
	multiply_variable = { ni_auth = 0.2 }
	add_to_variable = { eu_sh_8 = ni_auth }

	# Total share
	set_variable = { eu_sh_tot = 0 }
	add_to_variable = { eu_sh_tot = eu_sh_1 }
	add_to_variable = { eu_sh_tot = eu_sh_2 }
	add_to_variable = { eu_sh_tot = eu_sh_3 }
	add_to_variable = { eu_sh_tot = eu_sh_4 }
	add_to_variable = { eu_sh_tot = eu_sh_5 }
	add_to_variable = { eu_sh_tot = eu_sh_6 }
	add_to_variable = { eu_sh_tot = eu_sh_7 }
	add_to_variable = { eu_sh_tot = eu_sh_8 }
	if = { limit = { check_variable = { eu_sh_tot < 0.01 } } set_variable = { eu_sh_tot = 1.0 } }

	# If country has seats, distribute them proportionately
	set_variable = { my_seats = DVE_eu_ep_seats }
	if = { limit = { check_variable = { my_seats < 1 } } set_variable = { my_seats = 20 } }

	set_variable = { my_allocated = 0 }
	set_variable = { DVE_eu_ep_del_1 = eu_sh_1 }

# Council QMV Voting Calculation Effect
DVE_eu_qmv_check_effect = {
	set_variable = { global.DVE_eu_council_states_for = 0 }
	set_variable = { global.DVE_eu_council_states_against = 0 }
	set_variable = { global.DVE_eu_council_pop_for = 0 }
	set_variable = { global.DVE_eu_council_pop_against = 0 }
	set_variable = { global.DVE_eu_council_states_abstain = 0 }

	every_country = {
		limit = { has_country_flag = DVE_eu_member }
		if = {
			limit = { check_variable = { DVE_eu_council_vote = 2 } }
			add_to_variable = { global.DVE_eu_council_states_for = 1 }
			add_to_variable = { global.DVE_eu_council_pop_for = DVE_eu_council_weight }
		}
		else_if = {
			limit = { check_variable = { DVE_eu_council_vote = 3 } }
			add_to_variable = { global.DVE_eu_council_states_against = 1 }
			add_to_variable = { global.DVE_eu_council_pop_against = DVE_eu_council_weight }
		}
		else_if = {
			limit = { check_variable = { DVE_eu_council_vote = 1 } }
			add_to_variable = { global.DVE_eu_council_states_abstain = 1 }
		}
	}

	# Calculate Pop percentage (weight / 4480 * 100)
	set_variable = { global.DVE_eu_council_pop_for_pct = global.DVE_eu_council_pop_for }
	multiply_variable = { global.DVE_eu_council_pop_for_pct = 100 }
	divide_variable = { global.DVE_eu_council_pop_for_pct = 4480 }
	round_variable = global.DVE_eu_council_pop_for_pct

	set_variable = { global.DVE_eu_council_pop_against_pct = global.DVE_eu_council_pop_against }
	multiply_variable = { global.DVE_eu_council_pop_against_pct = 100 }
	divide_variable = { global.DVE_eu_council_pop_against_pct = 4480 }
	round_variable = global.DVE_eu_council_pop_against_pct

	# QMV Pass Check: states_for > 14 (i.e. >=15) AND pop_for > 2911 (i.e. >=65% of 4480)
	set_variable = { global.DVE_eu_council_qmv_pass = 0 }
	if = {
		limit = {
			check_variable = { global.DVE_eu_council_states_for > 14 }
			check_variable = { global.DVE_eu_council_pop_for > 2911 }
		}
		set_variable = { global.DVE_eu_council_qmv_pass = 1 }
	}

	# Blocking Minority Check: states_against > 3 (>=4) AND pop_against > 1568 (>35%)
	set_variable = { global.DVE_eu_council_blocking_minority = 0 }
	if = {
		limit = {
			check_variable = { global.DVE_eu_council_states_against > 3 }
			check_variable = { global.DVE_eu_council_pop_against > 1568 }
		}
		set_variable = { global.DVE_eu_council_blocking_minority = 1 }
	}

	# Refresh the Council dots of the parliament window
	DVE_eu_council_frames_effect = yes
}
