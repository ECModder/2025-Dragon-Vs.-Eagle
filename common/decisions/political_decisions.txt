#   Questo file SOSTITUISCE common/decisions/political_decisions.txt
#   del gioco base. L'originale era interamente costruito sulle vecchie
#   ideologie vanilla (fascism / democratic / communism / neutrality),
#   che in questa mod non esistono piu': le sue decisioni erano quindi
#   gia' non funzionanti e producevano errori nel log.
#
#   Le vecchie azioni ideologiche (Anti-Fascist / Anti-Democratic /
#   Anti-Communist / Anti-Neutrality Raids, i quattro Ban Party e le
#   quattro Press Censorship) sono state rimosse e rimpiazzate dai
#   quattro raid qui sotto, allineati ai 12 gruppi ideologici DvE.
#
#   "Improved Worker Conditions" NON e' qui: sta in
#   common/decisions/_generic_decisions.txt del gioco base e resta
#   invariato, come richiesto.
# ============================================================

political_actions = {

	# --- sinistra radicale ---------------------------------------
	anti_radicals_raids = {

		icon = generic_civil_support

		visible = {
			NOT = { has_government = totalitarian_socialism }
			NOT = { has_government = authoritarian_socialism }
			NOT = { has_government = revolutionary_socialism }
			NOT = { has_government = vanguard_socialism }
		}

		available = {
			OR = {
				totalitarian_socialism > 0.05
				authoritarian_socialism > 0.05
				revolutionary_socialism > 0.05
				vanguard_socialism > 0.05
			}
		}

		cost = 50
		days_re_enable = 90

		complete_effect = {
			add_stability = -0.10
			add_popularity = { ideology = totalitarian_socialism popularity = -0.04 }
			add_popularity = { ideology = authoritarian_socialism popularity = -0.03 }
			add_popularity = { ideology = revolutionary_socialism popularity = -0.03 }
			add_popularity = { ideology = vanguard_socialism popularity = -0.02 }
		}

		ai_will_do = {
			base = 1
		}
	}
