INT_uprising_annex_category = {

	INT_uprising_annex_puppet = {
		icon = GFX_decision_generic_form_nation
		allowed = { tag = INT }
		targets = { CMU SPA GDR PRS ENG }
		target_trigger = {
			FROM = {
				exists = yes
				is_subject_of = ROOT
			}
		}
		available = {
			has_political_power > 449
			FROM = { NOT = { is_at_war = yes } }
		}
		cost = 450
		fire_only_once = no
		complete_effect = {
			# Cores first, while FROM still exists and owns its
			# states - annex_country dissolves it right after.
			FROM = {
				every_owned_state = { add_core_of = ROOT }
			}
			annex_country = { target = FROM transfer_troops = yes }
		}
		ai_will_do = { factor = 20 }
	}
}
