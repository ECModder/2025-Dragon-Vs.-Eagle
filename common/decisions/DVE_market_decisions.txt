# ============================================================
#   DvE - Generic Decisions: Arms / Surplus / Black Market
#
#   Straightforward equipment purchases: pay PP, get the stated
#   amount of equipment after a delay. All three markets share a
#   single pool of 3 concurrent orders, tracked in the country
#   variable DVE_market_orders_active (initialised to 0 on_startup
#   in common/on_actions/DVE_on_actions.txt). Every purchase
#   decision here increments it and fires a delayed event
#   (events/DVE_market_events.txt) that delivers the equipment
#   25 days later (the middle of the 20-30 day range) and frees
#   the slot back up.
#
#   MBT and SP Artillery purchases both use the medium_tank_
#   equipment archetype: this mod's tank tech (common/units/
#   equipment/tank_chassis.txt) is the unmodified NSB modular
#   tank designer, which has no separate self-propelled-artillery
#   equipment type distinct from the regular tank line - flagged
#   to Matteo, fine as a placeholder until/unless a dedicated SPG
#   archetype gets built.
# ============================================================

DVE_arms_market_category = {

	DVE_arms_market_infantry_equipment = {
		icon = GFX_decision_generic_tank
		allowed = { always = yes }
		visible = { always = yes }
		available = { check_variable = { var = DVE_market_orders_active value = 3 compare = less_than } }
		cost = 125
		fire_only_once = no
		complete_effect = {
			add_to_variable = { DVE_market_orders_active = 1 }
