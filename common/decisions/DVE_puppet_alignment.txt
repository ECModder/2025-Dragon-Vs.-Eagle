# ============================================================
#   DvE - allineamento dei satelliti
#
#   Quando l'Italia o l'Unione Sovietica governano con un'ideologia
#   rivoluzionaria possono imporre ai propri puppet lo stesso
#   partito e lo stesso organo dirigente che hanno in patria.
#
#   La decisione e' MIRATA sui satelliti (target_array =
#   ROOT.subjects): ne scegli uno alla volta. Sparisce da sola per
#   i puppet gia' allineati, perche' il target_trigger chiede che
#   NON abbiano gia' quel governo.
#
#   target_root_trigger controlla l'ideologia del padrone: cosi'
#   l'Italia avanguardista vede solo l'opzione "Internationalism" e
#   quella rivoluzionaria solo "RCP", mai tutte e due insieme.
# ============================================================

DVE_puppet_alignment_category = {

	ITA_align_puppet_vanguard = {

		icon = GFX_decision_generic_political_address
		allowed = { tag = ITA }

		# ROOT.subjects prendeva solo i puppet veri. In partita i satelliti
		# stanno quasi sempre nella fazione del padrone, quindi si usa
		# ROOT.allies e si accettano entrambi i casi.
		target_array = ROOT.allies
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = ROOT }
				OR = {
					is_subject_of = ROOT
					is_in_faction_with = ROOT
				}
				NOT = { has_government = vanguard_socialism }
			}
		}
		target_root_trigger = {
			has_government = vanguard_socialism
		}

		available = { has_political_power > 74 }
		visible = { has_government = vanguard_socialism }

		cost = 75
		fire_only_once = no
		days_re_enable = 30

		complete_effect = {
			FROM = {
				set_party_name = { ideology = vanguard_socialism name = "Internationalism" long_name = "Internationalism - Communist Struggle" }
				create_country_leader = {
					name = "Worker's Councils"
					desc = "POLITICS_LEADER_DESC"
					picture = ""
					expire = "1965.1.1"
					ideology = leninist_internationalism
				}
				add_popularity = { ideology = vanguard_socialism popularity = 0.60 }
				set_politics = {
					ruling_party = vanguard_socialism
					elections_allowed = no
				}
				add_stability = -0.10
				country_event = { id = dve_puppet.1 hours = 6 }
			}
			add_political_power = -10
		}

		ai_will_do = { base = 40 }
	}
