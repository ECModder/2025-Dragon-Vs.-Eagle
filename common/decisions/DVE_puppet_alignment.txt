# ============================================================
#   DvE - allineamento dei satelliti
#
#   Quando l'Italia o l'Unione Sovietica governano con una delle
#   quattro ideologie socialiste, impongono ai propri satelliti lo
#   STESSO partito e lo stesso organo dirigente che hanno in patria.
#
#   PERCHE' QUATTRO E NON DUE
#   La prima versione copriva solo vanguard_socialism e
#   revolutionary_socialism. Ma l'albero italiano puo' portare al
#   potere anche totalitarian_socialism e authoritarian_socialism,
#   e in quei due casi non compariva nessuna decisione: il menu
#   c'era ed era vuoto. Ora c'e' una decisione per ciascuna delle
#   quattro, per entrambi i padroni.
#
#   NOTA SUL TARGETING
#   Nessun target_array: la decisione agisce su TUTTI i satelliti
#   in un colpo solo, puppet formali e membri di fazione insieme.
#   Con il targeting la lista restava vuota nei casi misti e
#   spariva l'intera categoria.
# ============================================================

	ITA_align_puppet_totalitarian = {

		icon = GFX_decision_generic_political_address
		allowed = { tag = ITA }

		visible = { has_government = totalitarian_socialism }

		available = {
			has_political_power > 149
			any_country = {
				NOT = { tag = ROOT }
					OR = {
						is_subject_of = ROOT
						is_in_faction_with = ROOT
					}
					NOT = { has_government = totalitarian_socialism }
			}
		}

		cost = 150
		fire_only_once = no
		days_re_enable = 60

		complete_effect = {
			every_country = {
				limit = {
				NOT = { tag = ROOT }
					OR = {
						is_subject_of = ROOT
						is_in_faction_with = ROOT
					}
					NOT = { has_government = totalitarian_socialism }
				}
				set_party_name = { ideology = totalitarian_socialism name = "PdRC, PCI" long_name = "Partito della Rifondazione Comunista, nPCI" }
				create_country_leader = {
					name = "Central Committee"
					desc = "POLITICS_LEADER_DESC"
					picture = ""
					expire = "1965.1.1"
					ideology = stalinism
				}
				add_popularity = { ideology = totalitarian_socialism popularity = 0.60 }
				set_politics = {
					ruling_party = totalitarian_socialism
					elections_allowed = no
				}
				add_stability = -0.10
				country_event = { id = dve_puppet.1 hours = 6 }
			}
		}

		ai_will_do = { base = 40 }
	}
