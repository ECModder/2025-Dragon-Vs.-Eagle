# ============================================================
#   DvE - allineamento dei satelliti
#
#   Quando l'Italia o l'Unione Sovietica governano con un'ideologia
#   rivoluzionaria impongono ai propri satelliti lo stesso partito
#   e lo stesso organo dirigente che hanno in patria.
#
#   NOTA SULLA SCELTA TECNICA
#   La prima versione era una decisione MIRATA (target_array).
#   Non funzionava: con target_array = ROOT.subjects restavano
#   fuori i satelliti che stanno in fazione ma non sono puppet
#   formali, e con ROOT.allies restavano fuori i puppet di un paese
#   che non ha fazione. In entrambi i casi la lista dei bersagli
#   era vuota e - con visible_when_empty = no - spariva l'intera
#   categoria dal menu.
#   Ora non c'e' targeting: la decisione agisce su TUTTI i
#   satelliti in un colpo solo, siano essi puppet o membri di
#   fazione. Cosi' la categoria e' sempre visibile e la decisione
#   e' semplicemente grigia finche' non hai satelliti da allineare.
# ============================================================

	ITA_align_puppet_vanguard = {

		icon = GFX_decision_generic_political_address
		allowed = { tag = ITA }

		visible = { has_government = vanguard_socialism }

		available = {
			has_political_power > 149
			any_country = {
				NOT = { tag = ROOT }

					is_subject_of = ROOT

		cost = 150
		fire_only_once = no

				limit = {

						is_subject_of = ROOT

				}
				set_party_name = { ideology = vanguard_socialism name = "Internationalism" long_name = "Internationalism - Communist Struggle" }

					name = "Worker's Councils"
					desc = "POLITICS_LEADER_DESC"
					picture = ""
					expire = "1965.1.1"
					ideology = leninist_internationalism
				}
				add_popularity = { ideology = vanguard_socialism popularity = 0.60 }
				set_politics = {
					ruling_party = vanguard_socialism
					elections_allowed = no
				}
				add_stability = -0.10
				country_event = { id = dve_puppet.1 hours = 6 }
			}
		}

		ai_will_do = { base = 40 }
	}
