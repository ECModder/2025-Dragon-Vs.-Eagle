# ============================================================
#   DvE - la guerra civile giorno per giorno
#
#   Menu comune alle quattro fazioni principali: RUS, SOV, RUR e
#   NRE. Si apre con la bandiera DVE_civil_war_active, che
#   RUS_second_civil_war_effect mette a tutte e quattro.
#
#   Tutto costa punti politici: in guerra civile il tempo del
#   governo e' la risorsa che scarseggia davvero.
#
#   La milizia usa il template "Territorialnoye Opolcheniye", che
#   esiste in tutti e quattro gli ordini di battaglia
#   (history/units/*_civil_war.txt e RUS_2025.txt).
# ============================================================

DVE_civil_war_category = {

	# ==========================================================
	#   MILIZIA TERRITORIALE
	#   Due livelli: la chiamata locale ogni 10 giorni, la leva di
	#   massa ogni 20. Non si possono usare a raffica.
	# ==========================================================
	DVE_cw_call_militia = {

		icon = GFX_decision_generic_army_support
		allowed = { OR = { tag = RUS tag = SOV tag = RUR tag = NRE } }

		available = {
			has_political_power > 49
			has_manpower > 4999
		}
		visible = { has_country_flag = DVE_civil_war_active }

		cost = 50
		fire_only_once = no
		days_re_enable = 10

		complete_effect = {
			create_unit = {
				division = "name = \"Opolcheniye\" division_template = \"Territorialnoye Opolcheniye\" start_experience_factor = 0.10 start_equipment_factor = 0.50"
				owner = ROOT
				count = 2
				allow_spawning_on_enemy_provs = no
			}
			add_manpower = -5000
			add_war_support = 0.02
		}

		ai_will_do = { base = 30 }
	}

	# ==========================================================
	#   CONSENSO
	#   Ogni fazione spinge le proprie ideologie. La decisione e'
	#   una sola: dentro sceglie in base al tag.
	# ==========================================================
	DVE_cw_rally_support = {

		icon = GFX_decision_generic_political_address
		allowed = { OR = { tag = RUS tag = SOV tag = RUR tag = NRE } }

		available = { has_political_power > 59 }
		visible = { has_country_flag = DVE_civil_war_active }

		cost = 60
		fire_only_once = no
		days_re_enable = 15

		complete_effect = {
			if = {
				limit = { tag = SOV }
				add_popularity = { ideology = totalitarian_socialism popularity = 0.03 }
				add_popularity = { ideology = authoritarian_socialism popularity = 0.03 }
				add_popularity = { ideology = vanguard_socialism popularity = 0.03 }
				add_popularity = { ideology = revolutionary_socialism popularity = 0.02 }
			}
			else_if = {
				limit = { tag = RUR }
				add_popularity = { ideology = progressive popularity = 0.04 }
				add_popularity = { ideology = centrist popularity = 0.04 }
				add_popularity = { ideology = conservative popularity = 0.03 }
			}
			else_if = {
				limit = { tag = RUS }
				add_popularity = { ideology = reactionary popularity = 0.04 }
				add_popularity = { ideology = authoritarianism popularity = 0.04 }
				add_popularity = { ideology = autocracy popularity = 0.03 }
			}
			else_if = {
				limit = { tag = NRE }
				add_popularity = { ideology = absolutism popularity = 0.04 }
				add_popularity = { ideology = ultranationalism popularity = 0.04 }
				add_popularity = { ideology = fundamentalism popularity = 0.03 }
			}
			add_stability = 0.03
		}

		ai_will_do = { base = 25 }
	}
