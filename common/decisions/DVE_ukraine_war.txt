# ============================================================
#   DvE - LA GUERRA D'UCRAINA: le due missioni a tempo
#
#   Stessa struttura della guerra italo-etiope del gioco base:
#   una missione con la barra del tempo residuo, che si chiude da
#   sola quando la guerra finisce e che, se scade, impone la pace.
#
#   278 giorni per entrambe. Gli obiettivi non si prendono con una
#   decisione: li controlla on_state_control_changed
#   (common/scripted_effects/DVE_ukraine_war.txt).
# ============================================================

RUS_ukraine_war_category = {

	# ==========================================================
	#   LA CAMPAGNA D'UCRAINA - 278 giorni
	#   Donbass -> Odessa -> Kiev
	# ==========================================================
	RUS_ukraine_campaign = {

		icon = GFX_decision_generic_army_support
		allowed = { original_tag = RUS }
		priority = 1000

		# si chiude da sola appena la guerra finisce, comunque finisca
		available = {
			has_country_flag = RUS_ukraine_war_over
		}

		visible = {
			has_country_flag = RUS_ukraine_war_active
			NOT = { has_country_flag = RUS_ukraine_war_over }
		}

		activation = {
			original_tag = RUS
			has_country_flag = RUS_ukraine_war_active
			NOT = { has_country_flag = RUS_ukraine_war_over }
		}

		is_good = no
		fire_only_once = yes
		selectable_mission = yes
		days_mission_timeout = 278

		# scaduto il tempo senza vittoria: pace bianca
		timeout_effect = {
			DVE_ukraine_white_peace_effect = yes
		}

		complete_effect = { }

		ai_will_do = { base = 0 }
	}
}


UKR_ukraine_war_category = {

	# ==========================================================
	#   LA CAMPAGNA DI LIBERAZIONE - 278 giorni
	#   Donbass -> Crimea -> portare la guerra in Russia
	# ==========================================================
	UKR_liberation_campaign = {

		icon = GFX_decision_generic_army_support
		allowed = { original_tag = UKR }
		priority = 1000

		available = {
			has_country_flag = UKR_ukraine_war_over
		}

		visible = {
			has_country_flag = UKR_ukraine_war_active
			NOT = { has_country_flag = UKR_ukraine_war_over }
		}

		activation = {
			original_tag = UKR
			has_country_flag = UKR_ukraine_war_active
			NOT = { has_country_flag = UKR_ukraine_war_over }
		}

		is_good = no
		fire_only_once = yes
		selectable_mission = yes
		days_mission_timeout = 278

		# la pace bianca la impone l'effetto condiviso, che e'
		# protetto da flag: se l'ha gia' fatta la missione russa
		# qui non succede niente
		timeout_effect = {
			RUS = { DVE_ukraine_white_peace_effect = yes }
		}

		complete_effect = { }

		ai_will_do = { base = 0 }
	}
}
