# DvE - European Union On-Actions
on_actions = {

	# Game Start: initialize all 27 EU member states, candidates, seats, and QMV weights
	on_startup = {
		effect = {
			if = {
				limit = { country_exists = GER }
				GER = { DVE_eu_init_effect = yes }
			}
		}
	}

	# Daily: refresh for human player
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				DVE_eu_refresh_effect = yes
			}
		}
	}

	# Weekly: recalculate EU state and presidency rotation
	on_weekly = {
		effect = {
			if = {
				limit = {
					OR = {
						has_country_flag = DVE_eu_member
						has_country_flag = DVE_eu_candidate
					}
				}
				DVE_eu_refresh_effect = yes
			}

			# Keep the European Parliament window up to date while it is open
			if = {
				limit = { has_country_flag = DVE_eu_parliament_open }
				DVE_eu_parliament_recalc_effect = yes
			}

			# Rotate Council presidency every 26 weeks (6 months)
			add_to_variable = { global.DVE_eu_presidency_counter = 1 }
			if = {
				limit = { check_variable = { global.DVE_eu_presidency_counter > 25 } }
				set_variable = { global.DVE_eu_presidency_counter = 0 }
				add_to_variable = { global.DVE_eu_council_presidency = 1 }
				if = {
					limit = { check_variable = { global.DVE_eu_council_presidency > 7 } }
					set_variable = { global.DVE_eu_council_presidency = 1 }
				}
			}
		}
	}
}
