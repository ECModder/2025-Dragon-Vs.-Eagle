# DvE - schedules the 2025 world news (events/DVE_news_2025.txt).
# Delays are hours from the 2025.01.01 12:00 start, so each news
# lands on its design date. The global flag makes it run once.
on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = DVE_news_2025_scheduled } }
				set_global_flag = DVE_news_2025_scheduled
				every_country = {
					news_event = { id = dve_news.13 hours = 1 }    # Somalia  - 01.01 01:00
					news_event = { id = dve_news.11 hours = 337 }  # Yemen    - 01.15 13:00
					news_event = { id = dve_news.1 hours = 414 }   # Gaza     - 01.18 18:00
					news_event = { id = dve_news.5 hours = 462 }   # Sahel    - 01.20 18:00
					news_event = { id = dve_news.12 hours = 524 }  # Tigray   - 01.23 08:00
					news_event = { id = dve_news.7 hours = 746 }   # Myanmar  - 02.01 14:00
				}
			}
		}
	}
}
