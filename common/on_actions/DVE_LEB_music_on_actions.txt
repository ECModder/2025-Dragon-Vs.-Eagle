# DvE - Lebanese civil war music. When the human Lebanon completes the
# focus that picks a side, the first song of that side starts at once, for
# that player only. The rest of the playlist is in the decisions
# (common/decisions/DVE_LEB_music.txt).
on_actions = {
	on_daily_LEB = {
		effect = {
			if = {
				limit = {
					is_ai = no
					has_completed_focus = LEB_Trust_the_current_government
					NOT = { has_country_flag = DVE_leb_music_government_played }
				}
				set_country_flag = DVE_leb_music_government_played
				scoped_play_song = "dve_leb_el_tarikh_rtajaf"
			}
			if = {
				limit = {
					is_ai = no
					has_completed_focus = LEB_Go_with_the_Islamic_resistance
					NOT = { has_country_flag = DVE_leb_music_resistance_played }
				}
				set_country_flag = DVE_leb_music_resistance_played
				scoped_play_song = "dve_leb_the_most_beautiful_story"
			}
		}
	}
}
